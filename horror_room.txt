local function chamberCenter(k)
    local col = k <= 5 and k or 11 - k
    return (col - 0.5) * CELL, (k <= 5 and 0.5 or -0.5) * CELL

-- Point in chamber k, given as an offset from the centre of its floor.

    local cx, cy = chamberCenter(k)
    return Vector(cx + dx, cy + dy, z or 0)
end

local function chamberAt(lp)
    if lp.z < -30 or lp.z > H + 30 then return 0 end
    if lp.x < 0 or lp.x >= LEN or lp.y <= -CELL or lp.y >= CELL then return 0 end
    local col = math.floor(lp.x / CELL) + 1
    if lp.y >= 0 then return col end
    return 11 - col
end

-- Task scheduler: choreography is written as coroutines that wait().
---------------------------------------------------------------------------
local tasks = {}

local function spawnTask(name, fn, onError)
    local t = { name = name, co = coroutine.create(fn), wake = 0, onError = onError }
    tasks[#tasks + 1] = t
    return t
end

local function wait(t) coroutine.yield(t or 0) end

local function waitUntil(pred, timeout)
    local limit = now + (timeout or 1e9)
    while not pred() do
        if now >= limit then return false end
        coroutine.yield(0.05)
    end
    return true
end

local function after(delay, fn)
    return spawnTask("after", function() wait(delay) fn() end)
end

local function runTasks()
    local i = 1
    while i <= #tasks do
        local t = tasks[i]
        if t.dead then
            table.remove(tasks, i)
        else
            if now >= t.wake then
                local ok, delay = pcall(coroutine.resume, t.co)
                if not ok then
                    t.dead = true
                    say("task '" .. t.name .. "' stopped: " .. tostring(delay))
                    if t.onError then pcall(t.onError) end
                elseif coroutine.status(t.co) == "dead" then
                    t.dead = true
                else
                    t.wake = now + (tonumber(delay) or 0)
                end
            end
            i = i + 1
        end
    end
end

local function killTasks()
    for _, t in ipairs(tasks) do
        if not t.keep then t.dead = true end
    end
end

-- Networking (queued so a burst of effects can never trip the net limit)

local outbox = {}
local function post(fn) outbox[#outbox + 1] = fn end

local function flushNet()
    local sent = 0
    while outbox[1] and sent < 10 do
        local okb, left = pcall(net.getBytesLeft)
        if okb and left and left < 600 then return end
        local fn = table.remove(outbox, 1)
        local ok, err = pcall(fn)
        if not ok then say("net: " .. tostring(err)) end
        sent = sent + 1
    end
end

local function validPlayers(list)
    if not list then return nil end
    local r = {}
    for _, p in ipairs(list) do
        if isValid(p) then r[#r + 1] = p end
    end
    return r
end

local persist = { objective = "", tension = 0, tvState = 0, tvText = "" }

local function fx(targets, kind, a, b, c, d, e)
    if targets and #targets == 0 then return end
    if not targets then
        if kind == FX.OBJ then persist.objective = a or "" end
        if kind == FX.TENSION then persist.tension = a or 0 end
        if kind == FX.TV then persist.tvState, persist.tvText = a or 0, b or "" end
    end
    post(function()
        local to = validPlayers(targets)
        if to and #to == 0 then return end
        net.start("hr_fx")
        net.writeUInt(kind, 5)
        if kind == FX.FLASH then
            net.writeUInt(a, 8) net.writeUInt(b, 8) net.writeUInt(c, 8) net.writeUInt(d, 8) net.writeFloat(e or 0.5)
        elseif kind == FX.SHAKE then
            net.writeFloat(a) net.writeFloat(b) net.writeFloat(c)
        elseif kind == FX.SUB then
            net.writeString(a) net.writeFloat(b or 4)
        elseif kind == FX.OBJ then
            net.writeString(a or "")
        elseif kind == FX.SCARE then
            net.writeUInt(a, 4) net.writeFloat(b or 0.7)
        elseif kind == FX.TENSION then
            net.writeUInt(math.floor(clamp(a or 0, 0, 1) * 255), 8)
        elseif kind == FX.TITLE then
            net.writeString(a) net.writeString(b or "") net.writeFloat(c or 6)
        elseif kind == FX.TV then
            net.writeUInt(a or 0, 2) net.writeString(b or "")
        end
        net.send(to)
    end)
end

-- Lamps -------------------------------------------------------------------
local lampState = {}

local function writeLamp(id)
    local s = lampState[id]
    net.writeUInt(s.mode, 4)
    net.writeUInt(s.r, 8) net.writeUInt(s.g, 8) net.writeUInt(s.b, 8)
    net.writeUInt(clamp(math.floor(s.level * 100 + 0.5), 0, 255), 8)
end

local function lampPost(id)
    local s = lampState[id]
    if s.dirty then return end
    s.dirty = true
    post(function()
        s.dirty = false
        net.start("hr_lamp")
        net.writeUInt(id, 6)
        writeLamp(id)
        net.send()
    end)
end

-- t may contain mode, level and col = { r, g, b }
local function lampSet(id, t)
    local s = lampState[id]
    if not s then return end
    if t.mode then s.mode = t.mode end
    if t.level then s.level = t.level end
    if t.col then s.r, s.g, s.b = t.col[1], t.col[2], t.col[3] end
    lampPost(id)
end

local function lampDefault(id)
    local d = LAMP_DEFAULT[id]
    local s = lampState[id] or {}
    lampState[id] = s
    s.mode, s.r, s.g, s.b, s.level = d[1], d[2], d[3], d[4], d[5]
    lampPost(id)
end

-- Interactables -----------------------------------------------------------
local intState = {}

local function intSet(id, active, handler)
    local s = intState[id]
    if not s then s = { active = false } intState[id] = s end
    if handler ~= nil then s.handler = handler end
    if s.active ~= active then
        s.active = active
        post(function()
            net.start("hr_int")
            net.writeUInt(id, 5)
            net.writeBool(s.active == true)
            net.send()
        end)
    end
end

local function sendSync(ply)
    post(function()
        if ply ~= nil and not isValid(ply) then return end
        net.start("hr_sync")
        net.writeFloat(ORIGIN.x) net.writeFloat(ORIGIN.y) net.writeFloat(ORIGIN.z)
        net.writeFloat(BASE.y)
        net.writeBool(ready)
        for id = 1, #LAMPS do writeLamp(id) end
        local mask = 0
        for id, s in pairs(intState) do
            if s.active then mask = mask + 2 ^ (id - 1) end
        end
        net.writeUInt(mask, 16)
        net.writeString(persist.objective)
        net.writeUInt(math.floor(clamp(persist.tension, 0, 1) * 255), 8)
        net.writeUInt(persist.tvState, 2)
        net.writeString(persist.tvText)
        net.send(ply)
    end)
end

local roster = {}
local byChamber = {}
for k = 0, 10 do byChamber[k] = {} end
local rosterAt, trackAt = 0, 0

local function refreshPlayers()
    if now >= rosterAt then
        rosterAt = now + 1
        local seen = {}
        for _, e in ipairs(roster) do seen[e.ply] = e end
        local list = find.allPlayers()
        roster = {}
        for i, p in ipairs(list) do
            roster[i] = seen[p] or { ply = p, k = 0, lpos = Vector(0, 0, 0) }
        end
    end
    if now < trackAt then return end
    trackAt = now + 0.05
    for k = 0, 10 do byChamber[k] = {} end
    for _, e in ipairs(roster) do
        local p = e.ply
        if isValid(p) and p:isAlive() then
            e.lpos = toLocal(p:getPos())
            e.k = chamberAt(e.lpos)
            local t = byChamber[e.k]
            t[#t + 1] = e
        else
            e.k = -1
        end
    end
end

local function occ(k) return byChamber[k] end

local function plysOf(list)
    local r = {}
    for i, e in ipairs(list) do r[i] = e.ply end
    return r
end

-- Everyone inside chambers a..b
local function crowd(a, b)
    local r = {}
    for k = a, b or a do
        for _, e in ipairs(byChamber[k]) do r[#r + 1] = e end
    end
    return r
end

local function insiders() return plysOf(crowd(1, 10)) end

local function nearest(lpos, list)
    local best, bestD
    for _, e in ipairs(list) do
        local d = e.lpos:getDistance(lpos)
        if not bestD or d < bestD then best, bestD = e, d end
    end
    return best, bestD
end

local function eyeOf(e) return toLocal(e.ply:getShootPos()) end
local function aimOf(e) return dirToLocal(e.ply:getAimVector()) end

local function sees(e, lpos, minDot)
    local to = lpos - eyeOf(e)
    local d = to:getLength()
    if d < 1 then return true end
    return aimOf(e):dot(to / d) > (minDot or 0.6)
end

local function yawTo(from, to)
    return math.deg(math.atan2(to.y - from.y, to.x - from.x))
end

    if spawned % 3 == 0 or cpuAverage() > cpuMax() * 0.3 then coroutine.yield(0) end
end

-- Decor is tinted down with the room so nothing glows in the dark.
local function dim(r, g, b, mul)
    local m = clamp(CONFIG.AMBIENT * 1.35 * (mul or 1), 0, 1)
    return Color(math.floor(r * m), math.floor(g * m), math.floor(b * m))
end

local function holoRaw(models, wpos, wang, scale)
    if isstring(models) then models = { models } end
    for _, m in ipairs(models) do
        if not hologram.canSpawn() then break end
        local ok, h = pcall(hologram.create, wpos, wang, m, scale)
        if ok and h then return h end
    end
    skipped = skipped + 1
    return nil
end

local function glow(h)
    if not h then return end
    pcall(h.setMaterial, h, "debug/debugdrawflat")
    pcall(h.suppressEngineLighting, h, true)
end

-- Unit-cube primitives: scale is the size in units.
local function prim(mdl, lpos, size, lang, col, mat)
    local wp, wa = toWorld(lpos, lang)
    local h = holoRaw(mdl, wp, wa, size)
    if not h then return nil end
    if col then h:setColor(col) end
    if mat then pcall(h.setMaterial, h, mat) end
    return h
end
local function box(lpos, size, lang, col, mat) return prim(BOX, lpos, size, lang, col, mat) end

-- A model placed by its origin (characters stand on their origin).
local function figure(models, lpos, lang, scale, col)
    local wp, wa = toWorld(lpos, lang)
    local h = holoRaw(models, wp, wa, scale and Vector(scale, scale, scale) or nil)
    if not h then return nil end
    if col then h:setColor(col) end
    return h
end

local function bounds(ent)
    local mins, maxs = ent:obbMins(), ent:obbMaxs()
    if (maxs - mins):getLength() < 1 then
        local ok, a, b = pcall(ent.getModelBounds, ent)
        if ok and a and b then mins, maxs = a, b end
    end
    return mins, maxs
end

-- Rests an entity's bounding box on a floor point, centred, turned to a local yaw.
local function seat(ent, lpos, lyaw, scale)
    scale = scale or 1
    local wp, wa = toWorld(lpos, Angle(0, lyaw or 0, 0))
    ent:setAngles(wa)
    local mins, maxs = bounds(ent)
    local c = (mins + maxs) * (0.5 * scale)
    local off = localToWorld(Vector(c.x, c.y, mins.z * scale), ANG0, VEC0, wa)
    ent:setPos(wp - off)
end

-- Furniture hologram standing on the floor.
local function furn(models, lpos, lyaw, scale, col)
    local wp = toWorld(lpos)
    local h = holoRaw(models, wp, ANG0, scale and Vector(scale, scale, scale) or nil)
    if not h then return nil end
    seat(h, lpos, lyaw, scale)
    h:setColor(col or dim(255, 255, 255))
    return h
end

-- Props can only be spawned a few per second; this waits for the budget (build task only).
local function propRaw(mdl, wpos, wang)
    local tries = 0
    while not prop.canSpawn() do
        tries = tries + 1
        if tries > 600 then return nil, "prop limit reached" end
        coroutine.yield(0.05)
    end
    local ok, ent = pcall(prop.create, wpos, wang, mdl, true)
    if not ok then return nil, tostring(ent) end
    pcall(ent.setDrawShadow, ent, false)
    return ent
end

local function solidFurn(mdl, lpos, lyaw, col)
    local ent, err = propRaw(mdl, toWorld(lpos + Vector(0, 0, 40)), ANG0)
    if not ent then return nil, err end
    seat(ent, lpos, lyaw)
    ent:setColor(col or dim(255, 255, 255))
    return ent
end

local AX = { x = Vector(1, 0, 0), y = Vector(0, 1, 0), z = Vector(0, 0, 1) }
local warnedPlate = false

-- Fits a PHX plate into a slab of the room. kind: "floor" (len1 along X, len2 along Y),
-- "wallx" (plane across X: len1 along Y, len2 up) or "wally" (plane across Y: len1 along X, len2 up).
-- The model's real bounding box decides its rotation, so axis conventions don't matter.
local function plate(mdl, kind, center, len1, len2, mat, col)
    local n, t1, t2
    if kind == "floor" then n, t1, t2 = AX.z, AX.x, AX.y
    elseif kind == "wallx" then n, t1, t2 = AX.x, AX.y, AX.z
    else n, t1, t2 = AX.y, AX.x, AX.z end
    local wn, w1, w2 = dirToWorld(n), dirToWorld(t1), dirToWorld(t2)
    local wc = toWorld(center)

    local ent, err = propRaw("models/hunter/plates/" .. mdl .. ".mdl", wc, ANG0)
    if not ent then return nil, err end

    local size = ent:obbSize()
    local fwd = (math.abs(size.x - len1) <= math.abs(size.x - len2)) and w1 or w2
    ent:setAngles(fwd:getAngleEx(wn))
    ent:setPos(ent:getPos() + (wc - ent:localToWorld(ent:obbCenter())))

    local long, short = math.max(size.x, size.y), math.min(size.x, size.y)
    if not warnedPlate and (math.abs(long - math.max(len1, len2)) > 4 or math.abs(short - math.min(len1, len2)) > 4) then
        warnedPlate = true
        say("warning: " .. mdl .. " is not the size the layout expects; walls may not line up.")
    end

    if mat then pcall(ent.setMaterial, ent, mat) end
    if col then ent:setColor(col) end
    return ent
end

-- Doors (slabs that rise into the ceiling)

local doors = {}

local function doorClear(i)
    local g = G[i]
    for _, e in ipairs(roster) do
        if e.k and e.k >= 0 and math.abs(e.lpos.x - g.x) < 44 and math.abs(e.lpos.y - g.y) < 64 and e.lpos.z < H then
            return false
        end
    end
    return true
end

local function doorMove(i, target, dur, slam)
    local d = doors[i]
    if not d or d.target == target then return end
    d.target, d.dur, d.slam = target, math.max(dur, 0.05), slam
    if not slam then sndOn(d.ent, "doors/heavy_metal_move1.wav", 80, target == 1 and 92 or 100, 0.9) end
end

local function doorOpen(i, dur) doorMove(i, 1, dur or 2.6) end
local function doorClose(i, dur, slam) doorMove(i, 0, dur or 1.6, slam) end

-- Slams a door shut, but never on top of somebody standing in the doorway (call from a task).
local function slamDoor(i)
    if not waitUntil(function() return doorClear(i) end, 3) then return false end
    doorClose(i, 0.15, true)
    return true
end

local function updateDoors()
    for i = 0, 10 do
        local d = doors[i]
        if d and d.t ~= d.target and isValid(d.ent) then
            local step = dt / d.dur
            if d.target > d.t then d.t = math.min(d.t + step, d.target) else d.t = math.max(d.t - step, d.target) end
            local s = d.t * d.t * (3 - 2 * d.t)
            d.ent:setPos(d.closed + UP * (s * (H - 1)))
            if d.t == d.target then
                if d.slam then
                    sndOn(d.ent, "doors/heavy_metal_stop1.wav", 95, 85, 1)
                    sndOn(d.ent, "physics/metal/metal_box_impact_hard2.wav", 95, 70, 1)
                else
                    sndOn(d.ent, "doors/heavy_metal_stop1.wav", 78, 100, 0.8)
                end
            end
        end
    end
end

local actors = {}

local function actor(models, scale, col)
    local h = figure(models, Vector(-200, 0, -400), ANG0, scale, col)
    if not h then return nil end
    h:setNoDraw(true)
    local a = { h = h, pos = Vector(0, 0, 0), yaw = 0, shown = false }
    actors[#actors + 1] = a
    return a
end

local function alive(a) return a ~= nil and isValid(a.h) end

local function actorPlace(a, lpos, yaw, lang)
    if not alive(a) then return end
    a.pos = lpos
    a.yaw = yaw or a.yaw
    local wp, wa = toWorld(lpos, lang or Angle(0, a.yaw, 0))
    a.h:setPos(wp)
    a.h:setAngles(wa)
end

local function actorShow(a, on)
    if not alive(a) or a.shown == on then return end
    a.shown = on
    a.h:setNoDraw(not on)
end

local function actorAnim(a, names, rate)
    if not alive(a) then return end
    for _, n in ipairs(names) do
        local ok, id = pcall(a.h.lookupSequence, a.h, n)
        if ok and id and id >= 0 then
            pcall(a.h.setAnimation, a.h, id, 0, rate or 1)
            return
        end
    end
end

-- Moves an actor in a straight line; call from a task. stop() may abort the walk.
local function actorWalk(a, to, speed, steps, stride, stop)
    if not alive(a) then return end
    local nextStep = 0
    while true do
        local d = to - a.pos
        local dist = d:getLength()
        if dist < 2 then break end
        local move = math.min(dist, speed * dt)
        actorPlace(a, a.pos + d * (move / dist), yawTo(a.pos, to))
        if steps and now >= nextStep then
            nextStep = now + (stride or 0.4)
            sndOn(a.h, pick(steps), 80, math.random(90, 105), 0.9)
        end
        if stop and stop() then return true end
        coroutine.yield(0)
    end
end

local loops = {}

local function loopStop(name, fade)
    local s = loops[name]
    if not s then return end
    loops[name] = nil
    pcall(s.stop, s, fade or 0)
    if fade and fade > 0 then
        after(fade + 0.2, function() pcall(s.destroy, s) end).keep = true
    else
        pcall(s.destroy, s)
    end
end

-- Anything that might be a looping wav goes through here so it can always be stopped.
local function loopStart(name, ent, path, vol, pitch, level)
    loopStop(name)
    if not isValid(ent) or not sound.canCreate() then return nil end
    local ok, s = pcall(sound.create, ent, path)
    if not ok or not s then return nil end
    loops[name] = s
    pcall(function()
        if level then s:setSoundLevel(level) end
        s:play()
        s:setVolume(V(vol))
        if pitch then s:setPitch(pitch) end
    end)
    return s
end

local function loopTune(name, vol, pitch, fade)
    local s = loops[name]
    if not s then return end
    pcall(function()
        if vol then s:setVolume(V(vol), fade or 0) end
        if pitch then s:setPitch(pitch, fade or 0) end
    end)
end

local function loopStopAll()
    for name in pairs(loops) do loopStop(name) end
end

local function series(fmt, a, b)
    local t = {}
    for i = a, b do t[#t + 1] = string.format(fmt, i) end
    return t
end

-- Wall levers: a back plate, an arm and a knob built from primitives.
local function leverSet(lv, theta)
    lv.theta = theta
    local dirAng = Angle(-theta, lv.yaw, 0)
    local d = dirAng:getForward()
    local p0 = lv.base + Angle(0, lv.yaw, 0):getForward() * 3
    setLocal(lv.arm, p0 + d * 11, dirAng)
    setLocal(lv.knob, p0 + d * 22, ANG0)
end

local function makeLever(base, yaw)
    local lv = { base = base, yaw = yaw }
    lv.plate = box(base, Vector(4, 14, 30), Angle(0, yaw, 0), dim(95, 95, 100), CONFIG.MAT.partition)
    lv.arm   = box(base, Vector(22, 2.4, 2.4), ANG0, dim(170, 170, 175))
    lv.knob  = prim(SPHERE, base, Vector(5.5, 5.5, 5.5), ANG0, dim(230, 40, 35, 1.6))
    leverSet(lv, 50)
    return lv
end

local function leverThrow(lv)
    local t0 = now
    while now - t0 < 0.3 do
        leverSet(lv, lerp(50, -50, (now - t0) / 0.3))
        coroutine.yield(0)
    end
    leverSet(lv, -50)
end

local function shadeCol(r, g, b, mul)
    local m = clamp(CONFIG.AMBIENT * (mul or 1), 0, 1)
    return Color(math.floor(r * m), math.floor(g * m), math.floor(b * m))
end

local function buildStructure()
    local M = CONFIG.MAT
    local floorC, wallC = shadeCol(205, 190, 170, 0.8), shadeCol(215, 205, 190)
    local ceilC, partC  = shadeCol(190, 185, 180, 1.15), shadeCol(205, 200, 190)
    local w, f, c, p = M.wall, M.floor, M.ceiling, M.partition
    local hz = H / 2
    local list = {
        -- floor and ceiling
        { "plate16x32", "floor", 16 * U, 0, -1.5,    32 * U, 16 * U, f, floorC },
        { "plate8x16",  "floor", 36 * U, 0, -1.5,    8 * U,  16 * U, f, floorC },
        { "plate16x32", "floor", 16 * U, 0, H + 1.5, 32 * U, 16 * U, c, ceilC },
        { "plate8x16",  "floor", 36 * U, 0, H + 1.5, 8 * U,  16 * U, c, ceilC },
        -- outer walls
        { "plate4x32", "wally", 16 * U, 8 * U,  hz, 32 * U, H, w, wallC },
        { "plate4x8",  "wally", 36 * U, 8 * U,  hz, 8 * U,  H, w, wallC },
        { "plate4x32", "wally", 16 * U, -8 * U, hz, 32 * U, H, w, wallC },
        { "plate4x8",  "wally", 36 * U, -8 * U, hz, 8 * U,  H, w, wallC },
        { "plate4x16", "wallx", 40 * U, 0,      hz, 16 * U, H, w, wallC },
        { "plate3x4",  "wallx", 0, -6.5 * U,    hz, 3 * U,  H, w, wallC },
        { "plate4x6",  "wallx", 0, 0,           hz, 6 * U,  H, w, wallC },
        { "plate3x4",  "wallx", 0, 6.5 * U,     hz, 3 * U,  H, w, wallC },
        -- spine between the two rows (open at the far end)
        { "plate4x32", "wally", 16 * U, 0, hz, 32 * U, H, w, wallC },
        { "plate4x6",  "wally", 35 * U, 0, hz, 6 * U,  H, w, wallC },
        -- partitions, each leaving a doorway on alternating sides
        { "plate4x6", "wallx", 8 * U,  3 * U,  hz, 6 * U, H, p, partC },
        { "plate4x6", "wallx", 16 * U, 5 * U,  hz, 6 * U, H, p, partC },
        { "plate4x6", "wallx", 24 * U, 3 * U,  hz, 6 * U, H, p, partC },
        { "plate4x6", "wallx", 32 * U, 5 * U,  hz, 6 * U, H, p, partC },
        { "plate4x6", "wallx", 32 * U, -3 * U, hz, 6 * U, H, p, partC },
        { "plate4x6", "wallx", 24 * U, -5 * U, hz, 6 * U, H, p, partC },
        { "plate4x6", "wallx", 16 * U, -3 * U, hz, 6 * U, H, p, partC },
        { "plate4x6", "wallx", 8 * U,  -5 * U, hz, 6 * U, H, p, partC },
    }
    for _, s in ipairs(list) do
        local ent, err = plate(s[1], s[2], Vector(s[3], s[4], s[5]), s[6], s[7], s[8], s[9])
        if not ent then return false, err end
    end

    -- interior walls that have to be solid
    plate("plate4x4", "wallx", P(7, -95, 100, hz), 4 * U, H, p, partC)                 -- end wall of the cells
    local bars, err = plate("plate4x6", "wally", P(7, 47.45, 10, hz), 6 * U, H)        -- invisible stop behind the bars
    if not bars then return false, err end
    bars:setColor(Color(255, 255, 255, 0))
    pcall(bars.setRenderMode, bars, RENDERMODE.TRANSCOLOR)
    plate("plate2x4", "wally", P(9, -142.5, -80, hz), 2 * U, H, p, partC)              -- hiding nook
    plate("plate1x4", "wallx", P(9, -95, -166.3, hz), 1 * U, H, p, partC)
    plate("plate4x6", "wallx", P(10, 63, 47.5, hz), 6 * U, H, p, partC)                -- final run
    plate("plate4x6", "wallx", P(10, -63, -47.5, hz), 6 * U, H, p, partC)

    local doorC = shadeCol(170, 150, 135, 0.9)
    for i = 0, 10 do
        local g = G[i]
        local ent, derr = plate("plate2x4", g.kind, Vector(g.x, g.y, hz), 2 * U, H, M.door, doorC)
        if not ent then return false, derr end
        doors[i] = { ent = ent, closed = ent:getPos(), t = 0, target = 0, dur = 1 }
    end
    return true
end

local function buildTerminal()
    -- a Starfall HUD component by the entrance: USE it to switch the screen effects on
    box(Vector(-46, 0, 21), Vector(7, 7, 42), ANG0, dim(100, 100, 106), CONFIG.MAT.partition)
    local tries = 0
    while not prop.canSpawn() and tries < 100 do tries = tries + 1 coroutine.yield(0.05) end
    local wp, wa = toWorld(Vector(-46, 0, 44), Angle(0, 180, 0))
    local ok, hud = pcall(prop.createComponent, wp, wa, "starfall_hud", "models/bull/dynamicbuttonsf.mdl", true)
    if ok and hud then
        E.hud = hud
        pcall(hud.linkComponent, hud, chip())
    end
end

local function groundCheck()
    local worst = 0
    local probes = { Vector(0, -CELL, 0), Vector(0, CELL, 0), Vector(LEN, -CELL, 0), Vector(LEN, CELL, 0), Vector(LEN / 2, 0, 0) }
    for _, c in ipairs(probes) do
        local wp = toWorld(c)
        local ok, tr = pcall(trace.line, wp + Vector(0, 0, 120), wp - Vector(0, 0, 500), nil, MASK.SOLID_BRUSHONLY)
        if ok and tr and tr.Hit then
            worst = math.max(worst, math.abs(tr.HitPos.z - (ORIGIN.z - 3.5)))
        else
            worst = 999
        end
    end
    if worst > 28 then
        say("warning: the ground ahead is not flat or open - the ward needs about 2000 x 800 units of level floor.")
    end
end

local function buildAll()
    groundCheck()
    local ok, err = buildStructure()
    if not ok then return false, err end
    buildCast()
    for k = 1, 10 do
        -- each room is dressed by its own task, so one that fails is reported instead of ending the build
        local done = false
        spawnTask("dressing room " .. k, function() decor[k]() done = true end, function() done = true end)
        waitUntil(function() return done end, 60)
    end
    buildTerminal()
    return true
end

local function twitch(m)
    if not m.head then
        local ok, id = pcall(m.h.lookupBone, m.h, "ValveBiped.Bip01_Head1")
        m.head = (ok and id and id >= 0) and id or -1
    end
    if m.head >= 0 then
        pcall(m.h.manipulateBoneAngles, m.h, m.head, Angle(math.random(-25, 25), math.random(-30, 30), math.random(-35, 35)))
    end
end

local function updateBodies()
    if not E.bodies then return end
    local prey = occ(5)
    for _, b in ipairs(E.bodies) do
        if isValid(b.h) then
            local t = now - b.t0
            if t > 2.2 then
                for _, e in ipairs(prey) do
                    local dx, dy = e.lpos.x - b.pivot.x, e.lpos.y - b.pivot.y
                    if dx * dx + dy * dy < 34 * 34 then
                        b.t0, b.amp, t = now, 18 + math.random(0, 10), 0
                        sndOn(b.h, pick(CHAINS), 70, math.random(85, 110), 0.8)
                        break
                    end
                end
            end
            if t < 7 then
                b.h:setAngles(angToWorld(Angle(180 + b.amp * math.exp(-t / 1.7) * math.sin(t * 4.4), b.yaw, 0)))
                b.moving = true
            elseif b.moving then
                b.moving = false
                b.h:setAngles(angToWorld(Angle(180, b.yaw, 0)))
            end
        end
    end
end

    for _, b in ipairs(E.bodies or {}) do

end

    end)

---------------------------------------------------------------------------
-- Lamps: a projected light (real falloff and shadows), a dynamic light and a bulb
---------------------------------------------------------------------------
local PT_GAIN, DL_GAIN = 3.0, 1.3
local lampFx = {}

local function lampValue(id, t)
    local s = lampState[id]
    if not s or s.mode == MODE.OFF then return 0 end
    local pat = PATTERN[s.mode] or "m"
    local n = #pat
    if n == 1 then return (string.byte(pat) - 97) / 12 end
    local x = t * 10 + lampFx[id].phase
    local i = math.floor(x)
    local a = (string.byte(pat, i % n + 1) - 97) / 12
    if SMOOTH[s.mode] then
        local b = (string.byte(pat, (i + 1) % n + 1) - 97) / 12
        return a + (b - a) * (x - i)
    end
    return a
end

-- Only the rooms next to the player are lit, which keeps the projected-light count tiny.
local function lampActive(L)
    if myK == 0 then
        return (L.chamber == 1 or L.chamber == 10) and myPos.x > -900 and myPos.x < 420 and math.abs(myPos.y) < 900
    end
    return math.abs(L.chamber - myK) <= 1
end

local function initLamps()
    for id, L in ipairs(LAMPS) do
        local f = { phase = math.random(0, 40), wpos = toWorld(L.pos), v = -1, bv = -1 }
        lampFx[id] = f
        if id <= 10 then
            local ok, h = pcall(hologram.create, f.wpos, Angle(), SPHERE, Vector(7, 7, 7))
            if ok and h then
                f.bulb = h
                pcall(h.setMaterial, h, "debug/debugdrawflat")
                pcall(h.suppressEngineLighting, h, true)
                h:setColor(Color(0, 0, 0))
            end
            -- the cord it hangs from
            local cp = toWorld(L.pos + Vector(0, 0, 9))
            local ok2, c = pcall(hologram.create, cp, Angle(), BOX, Vector(0.7, 0.7, 11))
            if ok2 and c then c:setColor(Color(12, 12, 12)) end
        end
    end
end

local function lampColor(s)
    if not s.col then s.col = Color(s.r, s.g, s.b) end
    return s.col
end

local function updateLamps()
    for id, L in ipairs(LAMPS) do
        local f, s = lampFx[id], lampState[id]
        local v = lampValue(id, now) * s.level * CONFIG.LAMP_BRIGHTNESS
        local act = lampActive(L)
        local col = lampColor(s)

        if L.proj then
            local want = act and s.mode ~= MODE.OFF
            if want and not f.pt and now >= (f.retry or 0) then
                local ok, pt = pcall(light.createProjected)
                if ok and pt then
                    f.pt, f.v, f.shadow, f.ptCol = pt, -1, nil, nil
                    pt:setTexture("effects/flashlight/soft")
                    pt:setPos(f.wpos)
                    pt:setAngles(Angle(90, BASE.y, 0))
                    pt:setFOV(L.fov)
                    pt:setNearZ(6)
                    pt:setFarZ(L.far)
                else
                    f.retry = now + 5
                end
            elseif not want and f.pt then
                pcall(f.pt.remove, f.pt)
                f.pt = nil
            end
            local pt = f.pt
            if pt then
                local dirty = false
                local shadow = CONFIG.LAMP_SHADOWS and id <= 10 and L.chamber == myK
                if shadow ~= f.shadow then
                    f.shadow = shadow
                    pt:setEnableShadows(shadow)
                    dirty = true
                end
                if L.spin then
                    pt:setAngles(Angle(18, BASE.y + now * L.spin, 0))
                    dirty = true
                end
                if v ~= f.v then
                    f.v = v
                    pt:setBrightness(v * PT_GAIN)
                    dirty = true
                end
                if f.ptCol ~= col then
                    f.ptCol = col
                    pt:setColor(col)
                    dirty = true
                end
                if dirty then pt:update() end
            end
        end

        if act and v > 0.02 then
            if not f.dl then
                local ok, dl = pcall(light.create, f.wpos, L.size, v * DL_GAIN, col)
                if ok and dl then f.dl = dl end
            end
            if f.dl then
                f.dl:setBrightness(v * DL_GAIN)
                f.dl:setColor(col)
                pcall(f.dl.draw, f.dl)
            end
        end

        if f.bulb and math.abs(v - f.bv) > 0.03 then
            f.bv = v
            local m = clamp(v, 0, 1)
            f.bulb:setColor(Color(s.r * m, s.g * m, s.b * m))
        end
    end
end

        render.createRenderTarget(rt)
        local m = material.create("UnlitGeneric")
        m:setTextureRenderTarget("$basetexture", rt)
        m:setInt("$flags", 128)         -- $additive: the black background adds nothing, the text glows faintly
        local h = hologram.create(chip():getPos(), Angle(), BOX, Vector(0.4, size, size))
        h:setMaterial("!" .. m:getName())
        pcall(h.setRenderGroup, h, RENDERGROUP.TRANSLUCENT)   -- additive panels must draw after the wall behind them
        h:setNoDraw(true)
        return h
    end
    text.write  = panel("ward13_write", 160)
    text.prompt = panel("ward13_prompt", 60)
    text.tv     = panel("ward13_tv", 26)
    text.fWrite  = render.createFont("Akbar", 170, 700, true)
    text.fWriteS = render.createFont("Akbar", 118, 700, true)
    text.fKey    = render.createFont("Roboto", 120, 900, true)
    text.fLabel  = render.createFont("Roboto", 70, 700, true)
    text.fTV     = render.createFont("Roboto", 150, 900, true)
    local wp, wa = toWorld(SPOT.tvScreen, Angle(0, 180, 0))
    text.tv:setPos(wp)
    text.tv:setAngles(wa)
    text.tv:setNoDraw(false)
    text.ok = true
end

local function drawWriting(w, seed)
    render.clear(Color(0, 0, 0, 255))
    local n = #w.lines
    local lh = 185
    local y0 = 512 - (n - 1) * lh / 2
    local bottom = y0
    for i, line in ipairs(w.lines) do
        render.setFont(text.fWrite)
        if render.getTextSize(line) > 960 then render.setFont(text.fWriteS) end
        local y = y0 + (i - 1) * lh
        render.setColor(Color(w.col[1] * 0.45, w.col[2] * 0.45, w.col[3] * 0.45))
        render.drawSimpleText(518, y + 7, line, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
        render.setColor(Color(w.col[1], w.col[2], w.col[3]))
        render.drawSimpleText(512, y, line, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
        bottom = y
    end
    -- runs of paint under the letters
    for i = 1, 12 do
        local x = 170 + (i * 197 + seed * 53) % 690
        local len = 30 + (i * 61 + seed * 17) % 190
        render.setColor(Color(w.col[1] * 0.8, w.col[2] * 0.8, w.col[3] * 0.8))
        render.drawRect(x, bottom + 34, 4 + i % 3, len)
    end
end

local function drawPrompt()
    render.clear(Color(0, 0, 0, 255))
    local label = text.promptLabel
    if label == "" then return end
    local hold = text.promptHold
    render.setFont(text.fKey)
    render.setColor(Color(235, 235, 225))
    render.drawSimpleText(512, 400, hold and "[ HOLD E ]" or "[ E ]", TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
    render.setFont(text.fLabel)
    render.setColor(Color(200, 200, 190))
    render.drawSimpleText(512, 520, label, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
    if hold and holdT > 0 then
        local f = clamp(holdT / hold, 0, 1)
        render.setColor(Color(70, 70, 70))
        render.drawRect(212, 600, 600, 26)
        render.setColor(Color(200, 30, 30))
        render.drawRect(212, 600, 600 * f, 26)
    end
end

local function drawTV()
    render.clear(Color(0, 0, 0, 255))
    if tv.state == 0 then return end
    for i = 1, 80 do
        local g = math.random(20, 150)
        render.setColor(Color(g, g, g + 12))
        render.drawRect(math.random(-100, 1024), math.random(0, 1024), math.random(20, 260), math.random(3, 14))
    end
    render.setColor(Color(120, 130, 140))
    render.drawRect(0, (now * 380) % 1024, 1024, 40)
    if tv.state == 2 and (now * 2.2) % 1 < 0.72 then
        local a, b = string.match(tv.text, "^(.-)%s+(%S+)$")
        if not a then a, b = tv.text, "" end
        render.setFont(text.fTV)
        render.setColor(Color(255, 255, 255))
        render.drawSimpleText(512 + math.random(-10, 10), 420, a, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
        render.drawSimpleText(512 + math.random(-10, 10), 610, b, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
    end
end

local function updateText(ft)
    if not text.ok then return end
    -- wall writing follows the room the player is in
    if myK ~= text.writeIdx then
        text.writeIdx = myK
        local w = WRITINGS[myK]
        if w then
            local wp, wa = toWorld(w.pos, Angle(0, w.yaw, 0))
            text.write:setPos(wp)
            text.write:setAngles(wa)
            text.write:setScale(Vector(0.4, w.size, w.size))
            text.write:setNoDraw(false)
            text.writeDirty = true
        else
            text.write:setNoDraw(true)
        end
    end

    -- the prompt hovers over the nearest thing that can be used, facing the player
    local best, bestD
    local eye = eyePos()
    if myK > 0 then
        local leye = toLocal(eye)
        for id, def in pairs(INTERACT) do
            if intActive[id] then
                local d = leye:getDistance(def.pos)
                if d < def.radius * 2.2 and (not bestD or d < bestD) then best, bestD = id, d end
            end
        end
    end
    if best then
        local def = INTERACT[best]
        if text.promptLabel ~= def.label then
            text.promptLabel, text.promptHold = def.label, def.hold
            text.promptDirty = true
        end
        local wp = toWorld(def.pos + Vector(0, 0, 22))
        local dir = eye - wp
        text.prompt:setPos(wp)
        text.prompt:setAngles(Angle(0, dir:getAngle().y, 0))
        if text.promptHidden ~= false then
            text.promptHidden = false
            text.prompt:setNoDraw(false)
        end
        if def.hold and bestD < def.radius and me:keyDown(IN_KEY.USE) then
            holdT = holdT + ft
            text.promptDirty = true
        elseif holdT > 0 then
            holdT = 0
            text.promptDirty = true
        end
    else
        holdT = 0
        if text.promptHidden ~= true then
            text.promptHidden = true
            text.prompt:setNoDraw(true)
        end
    end

    -- render targets can be lost; repaint the static ones now and then
    if now >= text.refresh then
        text.refresh = now + 4
        text.writeDirty = true
        text.promptDirty = true
    end
end

hook.add("renderoffscreen", "ward13_rt", function()
    if not text.ok then return end
    if text.writeDirty then
        text.writeDirty = false
        local w = WRITINGS[text.writeIdx]
        if w then
            render.selectRenderTarget("ward13_write")
            drawWriting(w, text.writeIdx)
            render.selectRenderTarget()
        end
    end
    if text.promptDirty then
        text.promptDirty = false
        render.selectRenderTarget("ward13_prompt")
        drawPrompt()
        render.selectRenderTarget()
    end
    if (tv.state > 0 or tv.shown ~= tv.state) and now >= text.tvAt and math.abs(myK - 2) <= 1 then
        text.tvAt = now + 0.045
        tv.shown = tv.state
        render.selectRenderTarget("ward13_tv")
        drawTV()
        render.selectRenderTarget()
    end
end)

---------------------------------------------------------------------------
-- The scare: a body pinned to the camera for a fraction of a second
---------------------------------------------------------------------------
local function updateScare()
    if now < scare.till then
        local def = SCARE[scare.idx] or SCARE[1]
        if not scare.holo then
            local ok, h = pcall(hologram.create, eyePos(), Angle(), def.model, Vector(1, 1, 1))
            if ok and h then scare.holo, scare.model = h, def.model end
        elseif scare.model ~= def.model then
            pcall(scare.holo.setModel, scare.holo, def.model)
            scare.model = def.model
        end
        local ep, ea = eyePos(), eyeAngles()
        local fwd = ea:getForward()
        local h = scare.holo
        if h then
            local ang = Angle(math.rand(-4, 4), ea.y + 180 + math.rand(-5, 5), math.rand(-6, 6))
            local head = localToWorld(def.head, ANG0, VEC0, ang)
            h:setPos(ep + fwd * (20 + math.rand(-2, 2)) - head)
            h:setAngles(ang)
            if scare.hidden then
                scare.hidden = false
                h:setNoDraw(false)
            end
        end
        -- a hard light so the face is the brightest thing on screen
        if not scare.dl then
            local ok, dl = pcall(light.create, ep, 220, 4, Color(255, 235, 220))
            if ok and dl then scare.dl = dl end
        end
        if scare.dl then
            scare.dl:setPos(ep + fwd * 4)
            scare.dl:setBrightness(math.random(2, 5))
            pcall(scare.dl.draw, scare.dl)
        end
    elseif scare.holo and not scare.hidden then
        scare.hidden = true
        scare.holo:setNoDraw(true)
    end
end

-- Ambience that follows the player: a low drone and a heartbeat
---------------------------------------------------------------------------
local amb = { at = 0 }

local function initSounds()
    local ok, s = pcall(sound.create, me, "ambient/atmosphere/drone4lp.wav")
    if ok and s then amb.bed = s end
    local ok2, s2 = pcall(sound.create, me, "player/heartbeat1.wav")
    if ok2 and s2 then amb.heart = s2 end
end

local function fade(s, key, want, vol, pitch)
    if not s then return end
    if want then
        if not amb[key] or not s:isPlaying() then
            amb[key] = true
            s:play()
        end
        s:setVolume(clamp(vol * CONFIG.VOLUME, 0, 1))
        if pitch then s:setPitch(pitch) end
    elseif amb[key] then
        amb[key] = false
        s:stop(0.5)
    end
end

local function updateSounds()
    if now < amb.at then return end
    amb.at = now + 0.1
    fade(amb.bed, "bedOn", inside > 0.02, inside * 0.55)
    local beat = clamp((tensionShown - 0.4) * 1.7, 0, 1)
    fade(amb.heart, "heartOn", inside > 0.5 and beat > 0.02, beat, 88 + tensionShown * 40)
end

---------------------------------------------------------------------------
-- HUD (only for players with the Starfall HUD enabled)
---------------------------------------------------------------------------
local hud = {}

local function initHud()
    hud.fSub    = render.createFont("Roboto", 30, 500, true)
    hud.fObj    = render.createFont("Roboto", 21, 700, true)
    hud.fTitle  = render.createFont("Akbar", 120, 700, true)
    hud.fTitle2 = render.createFont("Roboto", 34, 400, true)
    for _, d in ipairs({ "l", "r", "u", "d" }) do
        local ok, m = pcall(material.load, "vgui/gradient-" .. d)
        if ok and m then hud[d] = m end
    end
end

local function vignette(w, h, col, frac)
    render.setColor(col)
    if hud.l then render.setMaterial(hud.l) render.drawTexturedRect(0, 0, w * frac, h) end
    if hud.r then render.setMaterial(hud.r) render.drawTexturedRect(w * (1 - frac), 0, w * frac, h) end
    if hud.u then render.setMaterial(hud.u) render.drawTexturedRect(0, 0, w, h * frac) end
    if hud.d then render.setMaterial(hud.d) render.drawTexturedRect(0, h * (1 - frac), w, h * frac) end
    render.setMaterial()
end

local function shadowText(x, y, str, col, alpha)
    render.setColor(Color(0, 0, 0, alpha))
    render.drawSimpleText(x + 2, y + 2, str, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
    render.setColor(Color(col[1], col[2], col[3], alpha))
    render.drawSimpleText(x, y, str, TEXT_ALIGN.CENTER, TEXT_ALIGN.CENTER)
end

-- alpha envelope: quick in, slow out
local function envelope(t, dur, fin, fout)
    if t < 0 or t > dur then return 0 end
    return clamp(math.min(t / fin, (dur - t) / fout), 0, 1)
end

hook.add("drawhud", "ward13_hud", function()
    if not synced or not hud.fSub then return end
    local w, h = render.getResolution()
    local a = inside

    if a > 0.01 then
        render.setColor(Color(0, 0, 0, 255 * CONFIG.HUD_DARKNESS * a))
        render.drawRect(0, 0, w, h)
        vignette(w, h, Color(0, 0, 0, clamp(175 + tensionShown * 70, 0, 255) * a), 0.34)
        if tensionShown > 0.5 then
            local pulse = (math.sin(now * (4 + tensionShown * 5)) * 0.5 + 0.5) * (tensionShown - 0.5) * 2
            vignette(w, h, Color(120, 0, 0, 110 * pulse * a), 0.2)
        end
        if CONFIG.FILM_GRAIN then
            for i = 1, 44 do
                local g = math.random(0, 1) * 255
                render.setColor(Color(g, g, g, math.random(6, 22) * a))
                render.drawRectFast(math.random(0, w), math.random(0, h), math.random(1, 3), math.random(1, 3))
            end
            if math.random(1, 40) == 1 then
                render.setColor(Color(255, 255, 255, 14 * a))
                render.drawRectFast(math.random(0, w), 0, 1, h)
            end
        end
    end

    local ft = (now - flash.t0) / flash.dur
    if ft >= 0 and ft < 1 then
        render.setColor(Color(flash.r, flash.g, flash.b, flash.a * (1 - ft)))
        render.drawRect(0, 0, w, h)
    end
    if now < scare.till then
        vignette(w, h, Color(150, 0, 0, 210), 0.3)
    end

    if objectiveText ~= "" and a > 0.3 then
        render.setFont(hud.fObj)
        shadowText(w / 2, h * 0.06, string.upper(objectiveText), { 200, 195, 180 }, 200 * a)
    end

    local sa = envelope(now - subtitle.t0, subtitle.dur, 0.35, 0.8)
    if sa > 0 then
        render.setFont(hud.fSub)
        -- typed out, a few letters per frame
        local shown = string.sub(subtitle.text, 1, math.floor((now - subtitle.t0) * 45) + 1)
        shadowText(w / 2, h * 0.84, shown, { 225, 220, 205 }, 255 * sa)
    end

    local ta = envelope(now - title.t0, title.dur, 0.25, 1.2)
    if ta > 0 then
        render.setFont(hud.fTitle)
        shadowText(w / 2, h * 0.42, title.a, { 190, 20, 20 }, 255 * ta)
        if title.b ~= "" then
            render.setFont(hud.fTitle2)
            shadowText(w / 2, h * 0.42 + 90, title.b, { 220, 215, 200 }, 255 * ta)
        end
    end
end)

local fogOn = false
local function updateFog()
    local want = CONFIG.FOG and inside > 0.05
    if want == fogOn then return end
    fogOn = want
    if want then
        hook.add("setupworldfog", "ward13_fog", function()
            render.setFogStart(40)
            render.setFogEnd(780)
            render.setFogDensity(0.86 * inside)
            render.setFogColor(Color(3, 3, 5))
        end)
    else
        hook.remove("setupworldfog", "ward13_fog")
    end
end

-- The owner gets the HUD straight away; everyone else can accept the request (USE on the chip)
-- or press USE on the terminal by the entrance.
pcall(setupPermissionRequest, { "enablehud" },
    "WARD 13 would like to draw its screen effects: darkness, fog, film grain, camera shake and subtitles.", true)

local function tryHud()
    if me == owner() and not CONFIG.OWNER_AUTO_HUD then return end
    local ok, allowed = pcall(hasPermission, "enablehud")
    if ok and allowed then pcall(enableHud, nil, true) end
end

hook.add("permissionrequest", "ward13_perm", function()
    local ok, yes = pcall(permissionRequestSatisfied)
    if ok and yes then pcall(enableHud, nil, true) end
end)

---------------------------------------------------------------------------
-- Main loop
---------------------------------------------------------------------------
local broken = {}

-- One failing subsystem (a refused permission, a hit limit) must not take the rest down.
local function guard(name, fn, arg)
    if broken[name] then return end
    local ok, err = pcall(fn, arg)
    if not ok then
        broken[name] = true
        pcall(print, "[WARD 13] " .. name .. " disabled: " .. tostring(err))
    end
end

hook.add("think", "ward13_client", function()
    local t = timer.curtime()
    local ft = clamp(t - now, 0, 0.1)
    now = t
    if not synced then return end
    if not inited then
        inited = true
        guard("lamps", initLamps)
        guard("text", initText)
        guard("sounds", initSounds)
        guard("hud", initHud)
        tryHud()
    end

    myPos = toLocal(me:getPos())
    myK = chamberAt(myPos)
    inside = inside + clamp((myK > 0 and 1 or 0) - inside, -ft * 1.2, ft * 1.6)
    tensionShown = tensionShown + clamp(tension - tensionShown, -ft * 0.5, ft * 1.5)

    guard("lamps", updateLamps)
    guard("text", updateText, ft)
    guard("scare", updateScare)
    guard("sounds", updateSounds)
    guard("fog", updateFog)
end)

end -- CLIENT
