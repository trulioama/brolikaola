--[[
    APOCALYPSE - a slow, map-wide end of the world for StarfallEx.

    Place the chip anywhere under open sky and type  !apoc start  (or press USE on it).

    What happens, in order
       1  Dusk      - the sky sinks to a deep navy while three cloud decks roll in from one horizon
       2  Storm     - distant sheet lightning, then ground strikes, thunder, wind and rain that keep building
       3  The Face  - a glow gathers behind the clouds and an enormous head pushes through them.
                      It blinks, looks around the map, picks people out and stares at them, and laughs
       4  Tornado   - it draws breath, purses its lips and blows; where the breath lands a funnel comes
                      down from the cloud base, touches the ground and starts to wander, growing with
                      every further breath

    Everything is drawn with clientside holograms and custom meshes, so every player sees and hears
    it with no setup. Players with the Starfall HUD also get camera shake and lightning screen flashes
    (the owner automatically, others by pressing USE on the chip and accepting the request).

    Owner chat commands
       !apoc start    begin the event            !apoc stop     let the storm clear, then clean up
       !apoc reset    remove everything at once  !apoc status   stage, time and tornado strength
       !apoc skip     jump to the next stage     !apoc laugh / !apoc blow   make the face do it now
]]

local CONFIG = {
    AUTO_START         = false,  -- begin as soon as the chip is placed
    CHAT_PREFIX        = "!apoc",
    ADMINS_CAN_CONTROL = false,  -- let server admins use the chat commands too
    TIME_SCALE         = 1.0,    -- 2 = the build-up runs twice as fast
    END_TIME           = 30,     -- seconds the storm takes to clear after "stop"
    VOLUME             = 1.0,
    DARKNESS           = 0.68,   -- how dark the world gets, 0 - 0.85
    SKY_DISTANCE       = 9000,   -- radius of the storm sky around each player (lower it on maps with a short view distance)
    FACE_SIZE          = 1.0,
    FACE_MODEL         = "models/gman_high.mdl",
    EYE_TRACKING       = true,   -- the eyes move on their own; turn off if they look wrong with a custom FACE_MODEL
    QUALITY            = 1.0,    -- 0.5 = lighter, 1.5 = denser clouds, funnel and debris
    RAIN               = true,
    TORNADO_PHYSICS    = true,   -- the tornado throws the physics props this chip is allowed to touch
    HUD_EFFECTS        = true,   -- camera shake and screen flash for players with the Starfall HUD
}

local sin, cos, tan, atan2 = math.sin, math.cos, math.tan, math.atan2
local min, max, floor, sqrt, abs, exp, random = math.min, math.max, math.floor, math.sqrt, math.abs, math.exp, math.random
local pi, rad, deg = math.pi, math.rad, math.deg

local function clamp(x, a, b) return x < a and a or (x > b and b or x) end
local function lerp(a, b, t) return a + (b - a) * t end
local function ramp(x, a, b)
    x = clamp((x - a) / (b - a), 0, 1)
    return x * x * (3 - 2 * x)
end
local function approach(cur, target, rate, dt) return cur + (target - cur) * (1 - exp(-rate * dt)) end

-- Event timeline in seconds (divided by TIME_SCALE)
local T = { STORM = 45, STORM_PEAK = 170, RAIN = 95, FACE = 150, FACE_OUT = 198, LAUGH = 208, BLOW = 236 }
local BLOW_INHALE, BLOW_LENGTH, LAUGH_LENGTH = 3.3, 12, 9
local WORLD = 16395 -- MASK_SOLID_BRUSHONLY: traces that only hit the map

-- SERVER: the director. Owns the clock, lightning, the face's decisions and the tornado's path.

if SERVER then

    local S = { phase = 0, t0 = 0, tEnd = 0, seed = 1, faceYaw = 0, centre = Vector(0, 0, 0), radius = 4000, skyZ = 4000 }
    local D = { tick = 0 }
    local tor, lastFace

    local function say(msg) print("[apocalypse] " .. msg) end
    local function traceWorld(a, b) return trace.line(a, b, nil, WORLD) end
    local function at(te) return S.t0 + te / CONFIG.TIME_SCALE end

    local function alivePlayers()
        local out = {}
        for _, p in ipairs(find.allPlayers()) do
            if p:isAlive() then out[#out + 1] = p end
        end
        return out
    end

    -- Finds open sky, the ground below it and roughly how far the playable area reaches.
    local function survey()
        local cands = { chip():getPos() + Vector(0, 0, 40) }
        for _, p in ipairs(alivePlayers()) do cands[#cands + 1] = p:getPos() + Vector(0, 0, 60) end
        local origin, skyZ, outdoor = cands[1], cands[1].z + 5000, false
        for i, c in ipairs(cands) do
            local tr = traceWorld(c, c + Vector(0, 0, 32000))
            if not tr.StartSolid then
                if tr.HitSky then
                    origin, skyZ, outdoor = c, tr.HitPos.z, true
                    break
                elseif i == 1 and tr.Hit then
                    skyZ = tr.HitPos.z
                end
            end
        end
        local down = traceWorld(origin, origin - Vector(0, 0, 16000))
        local groundZ = down.Hit and down.HitPos.z or origin.z
        local probeZ = max(min(groundZ + 420, skyZ - 48), groundZ + 16)

        local function ring(cx, cy)
            local dists, sx, sy = {}, 0, 0
            local from = Vector(cx, cy, probeZ)
            for i = 0, 15 do
                local a = i * pi / 8
                local tr = traceWorld(from, from + Vector(cos(a) * 15000, sin(a) * 15000, 0))
                local d = (tr.Fraction or 1) * 15000
                dists[#dists + 1] = d
                sx, sy = sx + cos(a) * d, sy + sin(a) * d
            end
            table.sort(dists)
            return dists[10], cx + sx / 16, cy + sy / 16
        end

        local cx, cy = origin.x, origin.y
        local reach, mx, my = ring(cx, cy)
        -- nudge the centre towards the middle of the open area when the way there is clear
        local mid = Vector(lerp(cx, mx, 0.7), lerp(cy, my, 0.7), probeZ)
        if not traceWorld(Vector(cx, cy, probeZ), mid).Hit then
            cx, cy = mid.x, mid.y
            reach = ring(cx, cy)
            local d2 = traceWorld(mid, mid - Vector(0, 0, 16000))
            if d2.Hit then groundZ = d2.HitPos.z end
        end
        S.centre = Vector(cx, cy, groundZ)
        S.radius = clamp(reach, 1500, 12000)
        S.skyZ = skyZ
        return outdoor
    end

    local function groundAt(x, y, zHint)
        local top = min(S.skyZ - 16, zHint + 700)
        local tr = traceWorld(Vector(x, y, top), Vector(x, y, top - 6000))
        if tr.StartSolid then
            tr = traceWorld(Vector(x, y, zHint + 120), Vector(x, y, zHint - 3000))
            if tr.StartSolid then return nil end
        end
        if tr.Hit and not tr.HitSky then return tr.HitPos.z end
        return nil
    end

    local function sendState(to)
        net.start("apoc_state")
        net.writeUInt(S.phase, 2)
        net.writeDouble(S.t0)
        net.writeDouble(S.tEnd)
        net.writeUInt(S.seed, 24)
        net.writeFloat(S.faceYaw)
        net.writeVector(S.centre)
        net.writeFloat(S.radius)
        net.send(to)
    end

    local function sendFace(a, to)
        net.start("apoc_face")
        net.writeUInt(a.kind, 3)
        net.writeDouble(a.t0)
        net.writeFloat(a.dur)
        net.writeVector(a.target)
        net.writeUInt(a.ent, 13)
        net.writeUInt(a.seed, 16)
        net.send(to)
    end

    -- kind: 1 look, 2 stare, 3 laugh, 4 blow
    local function faceAction(kind, dur, target, ent)
        lastFace = {
            kind = kind, t0 = timer.curtime(), dur = dur, target = target or S.centre,
            ent = ent and ent:entIndex() or 0, seed = random(0, 65535),
        }
        sendFace(lastFace)
    end

    local function sendTornado(now)
        net.start("apoc_tor")
        net.writeDouble(now)
        net.writeFloat(tor.x)
        net.writeFloat(tor.y)
        net.writeFloat(tor.z)
        net.writeFloat(tor.vx)
        net.writeFloat(tor.vy)
        net.writeFloat(tor.I)
        net.writeDouble(tor.born)
        net.send()
    end

    local function strike(storm)
        local power = clamp(0.25 + storm * 0.5 + random() * 0.4, 0, 1)
        local kind, v = 0, nil
        if random() < 0.1 + 0.32 * storm then
            local x, y
            local plys = alivePlayers()
            if storm > 0.5 and #plys > 0 and random() < 0.35 then
                local p = plys[random(#plys)]:getPos()
                local a, r = random() * 2 * pi, 500 + random() * 2300
                x, y = p.x + cos(a) * r, p.y + sin(a) * r
            else
                local a, r = random() * 2 * pi, sqrt(random()) * S.radius * 1.1
                x, y = S.centre.x + cos(a) * r, S.centre.y + sin(a) * r
            end
            local z = groundAt(x, y, S.centre.z)
            if z then kind, v = 1, Vector(x, y, z) end
        end
        if not v then
            -- sheet lightning inside the clouds: azimuth, elevation. Early flashes stay near the horizon.
            v = Vector(random() * 360, lerp(5, 58, random() ^ (3 - 2 * storm)), 0)
        end
        net.start("apoc_bolt")
        net.writeUInt(kind, 2)
        net.writeVector(v)
        net.writeUInt(random(0, 65535), 16)
        net.writeFloat(power)
        net.send()
    end

    local function randomPoint(scale)
        local a, r = random() * 2 * pi, sqrt(random()) * S.radius * scale
        return S.centre.x + cos(a) * r, S.centre.y + sin(a) * r
    end

    -- Where the first breath lands: inside the map, close enough to be seen, not on top of anyone.
    local function pickTouchdown()
        local plys = alivePlayers()
        local best, bestScore
        for _ = 1, 12 do
            local x, y = randomPoint(0.75)
            local nearest = 1e9
            for _, p in ipairs(plys) do
                local pp = p:getPos()
                nearest = min(nearest, sqrt((pp.x - x) ^ 2 + (pp.y - y) ^ 2))
            end
            local score = abs(nearest - 2200)
            local z = groundAt(x, y, S.centre.z)
            if z and (not bestScore or score < bestScore) then best, bestScore = Vector(x, y, z), score end
        end
        return best or Vector(S.centre.x, S.centre.y, S.centre.z)
    end

    local function spawnTornado(pos, now)
        tor = {
            x = pos.x, y = pos.y, z = pos.z, vx = 0, vy = 0, I = 0, boost = 0, born = now,
            heading = random() * 2 * pi, px = 0, py = 0, phase = random() * 6, wpUntil = 0, send = 0, phys = 0,
        }
    end

    local function retarget(now, x, y)
        if not x then
            local plys = alivePlayers()
            if #plys > 0 and random() < 0.45 then
                local p = plys[random(#plys)]:getPos()
                local a, r = random() * 2 * pi, random() * 1200
                x, y = p.x + cos(a) * r, p.y + sin(a) * r
            else
                x, y = randomPoint(0.8)
            end
        end
        local dx, dy = x - S.centre.x, y - S.centre.y
        local d = sqrt(dx * dx + dy * dy)
        local lim = S.radius * 0.9
        if d > lim then x, y = S.centre.x + dx / d * lim, S.centre.y + dy / d * lim end
        tor.wx, tor.wy, tor.wpUntil = x, y, now + 18 + random() * 22
    end

    local denied = {}
    local function throwProps(I)
        local reach = lerp(500, 2200, I)
        local ok, list = pcall(find.inSphere, Vector(tor.x, tor.y, tor.z + reach * 0.4), reach)
        if not ok or not list then return end
        local done = 0
        for _, ent in ipairs(list) do
            if done >= 40 then break end
            local idx = ent:entIndex()
            if not denied[idx] and string.sub(ent:getClass(), 1, 12) == "prop_physics" then
                done = done + 1
                local fine = pcall(function()
                    local ph = ent:getPhysicsObject()
                    if not ph:isValid() then return end
                    local p = ent:getPos()
                    local dx, dy = p.x - tor.x, p.y - tor.y
                    local d = sqrt(dx * dx + dy * dy) + 1
                    local fall = clamp(1 - d / reach, 0, 1)
                    local swirl, pull, lift = 1400 * fall * (0.3 + I), 700 * fall, 900 * fall * (0.4 + I)
                    local k = ph:getMass() * 0.2
                    ph:wake()
                    ph:applyForceCenter(Vector((-dy / d * swirl - dx / d * pull) * k, (dx / d * swirl - dy / d * pull) * k, lift * k))
                end)
                if not fine then denied[idx] = true end
            end
        end
    end

    local function updateTornado(now, dt)
        local age = now - tor.born
        local formed = age > 22
        local target = 0
        if S.phase == 1 then target = min(1, 0.72 * (1 - exp(-max(age - 22, 0) / 150)) + 0.08 + tor.boost) end
        tor.I = approach(tor.I, target, S.phase == 1 and 0.25 or 0.12, dt)
        if formed then
            local dw = tor.wx and sqrt((tor.wx - tor.x) ^ 2 + (tor.wy - tor.y) ^ 2) or 0
            if not tor.wx or now > tor.wpUntil or dw < 400 then retarget(now) end
            local want = atan2(tor.wy - tor.y, tor.wx - tor.x) + sin(now * 0.23 + tor.phase) * 0.6
            local diff = (want - tor.heading + pi) % (2 * pi) - pi
            local turn = lerp(0.36, 0.18, tor.I) * dt
            tor.heading = tor.heading + clamp(diff, -turn, turn)
            local speed = lerp(110, 360, tor.I) * (0.85 + 0.15 * sin(now * 0.41 + tor.phase * 2))
            tor.vx = approach(tor.vx, cos(tor.heading) * speed + tor.px, 0.8, dt)
            tor.vy = approach(tor.vy, sin(tor.heading) * speed + tor.py, 0.8, dt)
        else
            tor.vx, tor.vy = approach(tor.vx, tor.px, 0.8, dt), approach(tor.vy, tor.py, 0.8, dt)
        end
        tor.px, tor.py = approach(tor.px, 0, 0.35, dt), approach(tor.py, 0, 0.35, dt)
        tor.x, tor.y = tor.x + tor.vx * dt, tor.y + tor.vy * dt
        -- never leave the surveyed area
        local dx, dy = tor.x - S.centre.x, tor.y - S.centre.y
        local d = sqrt(dx * dx + dy * dy)
        if d > S.radius then
            tor.x, tor.y = S.centre.x + dx / d * S.radius, S.centre.y + dy / d * S.radius
            retarget(now, S.centre.x, S.centre.y)
        end
        local gz = groundAt(tor.x, tor.y, tor.z)
        if gz then tor.z = approach(tor.z, gz, 2.5, dt) end
        if now >= tor.send then
            tor.send = now + 0.2
            sendTornado(now)
        end
        if CONFIG.TORNADO_PHYSICS and formed and tor.I > 0.05 and now >= tor.phys then
            tor.phys = now + 0.2
            throwProps(tor.I)
        end
    end

    local function resetEvent()
        S.phase, S.tEnd = 0, 0
        tor, lastFace = nil, nil
        sendState()
    end

    local function startEvent()
        if S.phase ~= 0 then return false end
        local outdoor = survey()
        local now = timer.curtime()
        S.phase, S.t0, S.tEnd = 1, now, 0
        S.seed = random(1, 16000000)
        S.faceYaw = random() * 360
        tor, lastFace = nil, nil
        D = { tick = 0, bolt = 0, busy = 0, blowHit = nil }
        D.look, D.stare, D.laugh, D.blow = at(T.FACE + 30), at(T.FACE_OUT), at(T.LAUGH), at(T.BLOW)
        sendState()
        if not outdoor then say("no open sky found above the chip or any player - the storm needs sky to be seen.") end
        return true
    end

    local function stopEvent()
        if S.phase ~= 1 then return false end
        S.phase, S.tEnd = 2, timer.curtime()
        sendState()
        return true
    end

    local function direct(now)
        local plys = alivePlayers()
        if now >= D.blow then
            local target = tor and Vector(tor.x, tor.y, tor.z) or pickTouchdown()
            faceAction(4, BLOW_LENGTH, target)
            D.busy, D.blow = now + BLOW_LENGTH + 0.5, now + 70 + random() * 40
            D.blowHit, D.blowTarget = now + BLOW_INHALE + 0.4, target
        elseif now >= D.laugh then
            faceAction(3, LAUGH_LENGTH, nil, #plys > 0 and plys[random(#plys)] or nil)
            D.busy, D.laugh = now + LAUGH_LENGTH + 0.5, now + 45 + random() * 30
        elseif now >= D.stare and #plys > 0 then
            local p = plys[random(#plys)]
            local dur = 6 + random() * 3
            faceAction(2, dur, p:getPos(), p)
            D.busy, D.stare = now + dur + 0.5, now + 28 + random() * 17
        elseif now >= D.look then
            local dur = 3.5 + random() * 4.5
            local target
            if #plys > 0 and random() < 0.35 then
                target = plys[random(#plys)]:getPos()
            else
                local x, y = randomPoint(1)
                target = Vector(x, y, S.centre.z)
            end
            faceAction(1, dur, target)
            D.look = now + dur
        end
    end

    hook.add("think", "apoc_director", function()
        if S.phase == 0 then return end
        local now = timer.curtime()
        if now < D.tick then return end
        local dt = clamp(now - D.tick + 0.1, 0.05, 0.3)
        D.tick = now + 0.1
        if S.phase == 2 and now > S.tEnd + CONFIG.END_TIME + 1 then
            resetEvent()
            say("the sky is clear again.")
            return
        end
        local t = (now - S.t0) * CONFIG.TIME_SCALE
        local storm = ramp(t, T.STORM, T.STORM_PEAK)
        if S.phase == 2 then storm = storm * (1 - ramp(now - S.tEnd, 0, CONFIG.END_TIME * 0.6)) end
        if storm > 0.01 and now >= D.bolt then
            if D.bolt > 0 then strike(storm) end
            D.bolt = now + lerp(13, 2.4, storm) * (0.5 + random())
        end
        if S.phase == 1 then
            if D.blowHit and now >= D.blowHit then
                D.blowHit = nil
                if not tor then
                    spawnTornado(D.blowTarget, now)
                else
                    -- every further breath feeds the funnel and shoves it downwind
                    tor.boost = min(tor.boost + 0.14, 0.5)
                    local a = rad(S.faceYaw + 180 + (random() - 0.5) * 100)
                    tor.px, tor.py = cos(a) * 320, sin(a) * 320
                    tor.heading = a
                    retarget(now, tor.x + cos(a) * 3000, tor.y + sin(a) * 3000)
                end
            end
            if now >= D.busy and now >= at(T.FACE + 30) then direct(now) end
        end
        if tor then updateTornado(now, dt) end
    end)

    hook.add("ClientInitialized", "apoc_sync", function(ply)
        sendState(ply)
        if lastFace and timer.curtime() < lastFace.t0 + lastFace.dur then sendFace(lastFace, ply) end
    end)

    local function allowed(ply)
        return ply == owner() or (CONFIG.ADMINS_CAN_CONTROL and ply:isAdmin())
    end

    local STAGES = { { T.STORM, "storm" }, { T.RAIN, "rain" }, { T.FACE, "the face" }, { T.LAUGH, "laughter" }, { T.BLOW, "the tornado" } }

    hook.add("PlayerSay", "apoc_cmd", function(ply, text)
        if not isstring(text) or not allowed(ply) then return end
        local msg = string.lower(string.trim(text))
        local prefix = CONFIG.CHAT_PREFIX
        if string.sub(msg, 1, #prefix) ~= prefix then return end
        local arg = string.trim(string.sub(msg, #prefix + 1))
        local now = timer.curtime()
        if arg == "start" then
            say(startEvent() and "it begins. Watch the sky." or "already running - use stop or reset first.")
        elseif arg == "stop" then
            say(stopEvent() and ("the storm is breaking up, clear in " .. CONFIG.END_TIME .. " seconds.") or "nothing to stop.")
        elseif arg == "reset" then
            resetEvent()
            say("everything removed.")
        elseif arg == "status" then
            if S.phase == 0 then
                say("idle.")
            else
                local t = (now - S.t0) * CONFIG.TIME_SCALE
                local stage = "dusk"
                for _, s in ipairs(STAGES) do if t >= s[1] then stage = s[2] end end
                say(string.format("%s | stage: %s | %ds in | tornado: %s", S.phase == 1 and "running" or "ending", stage, floor(t),
                    tor and (floor(tor.I * 100) .. "%") or "none"))
            end
        elseif arg == "skip" then
            if S.phase ~= 1 then say("not running.") return end
            local t = (now - S.t0) * CONFIG.TIME_SCALE
            for _, s in ipairs(STAGES) do
                if s[1] > t + 1 then
                    local shift = (s[1] - t) / CONFIG.TIME_SCALE
                    S.t0 = S.t0 - shift
                    D.look, D.stare, D.laugh, D.blow = D.look - shift, D.stare - shift, D.laugh - shift, D.blow - shift
                    sendState()
                    say("skipped to " .. s[2] .. ".")
                    return
                end
            end
            say("already at the last stage.")
        elseif arg == "laugh" or arg == "blow" then
            if S.phase ~= 1 or now < at(T.FACE_OUT - 10) then say("the face is not out yet.") return end
            if arg == "laugh" then D.laugh = now else D.blow = now end
            D.busy = min(D.busy, now)
        else
            say("commands: " .. prefix .. " start | stop | reset | status | skip | laugh | blow")
        end
    end)

    hook.add("starfallused", "apoc_use", function(activator, used)
        if used == chip() and activator == owner() and S.phase == 0 then
            startEvent()
            say("it begins. Watch the sky.")
        end
    end)

    if CONFIG.AUTO_START then timer.simple(2, startEvent) end

    local function RNG(seed)
        local s = (seed % 2147483646) + 1
        return function(a, b)
            s = (s * 16807) % 2147483647
            local r = (s - 1) / 2147483646
            if a then return a + (b - a) * r end
            return r
        end
    end

    -- A failing part switches itself off instead of taking the whole chip down.
    local function guard(name, fn, ...)
        if broken[name] then return end
        local ok, err = pcall(fn, ...)
        if not ok then
            broken[name] = true
            if me == owner() then print("[apocalypse] " .. name .. " switched off: " .. tostring(err)) end
        end
    end

    local function newMat(tex, flags)
        local m = material.create("UnlitGeneric")
        m:setInt("$flags", flags)
        m:setTexture("$basetexture", tex)
        m:recompute()
        Res.mats[#Res.mats + 1] = m
        return m
    end

    local function findTexture(list)
        for _, path in ipairs(list) do
            local ok, name = pcall(material.getTexture, path, "$basetexture")
            if ok and isstring(name) and name ~= "" and not string.find(string.lower(name), "error", 1, true) then return name end
        end
    end

    local function newMesh(verts)
        local m = mesh.createFromTable(verts)
        Res.meshes[#Res.meshes + 1] = m
        return m
    end

    local function newHolo(model, pos)
        local h = hologram.create(pos or View.pos, Angle(0, 0, 0), model or BASE_MODEL)
        Res.holos[#Res.holos + 1] = h
        pcall(h.addEffects, h, 16) -- EF_NOSHADOW

    local function meshHolo(meshObj, mat, extent)
        local h = newHolo()
        h:setMesh(meshObj)
        h:setMeshMaterial(mat)
        h:setRenderGroup(RENDERGROUP.TRANSLUCENT)
        h:setRenderBounds(Vector(-extent, -extent, -extent), Vector(extent, extent, extent))
        h:setNoDraw(true)
        return h
    end

    -- A set of holograms sharing one material. Colour and alpha are written to both the entity and
    -- the material so the result is the same whichever of the two the mesh draw honours.
    local function drawable(holos, mat, manual)
        return { hs = holos, m = mat, key = 0, off = true, manual = manual }
    end

    local function tint(d, r, g, b, a)
        local r8, g8, b8 = floor(clamp(r, 0, 1) * 255 + 0.5), floor(clamp(g, 0, 1) * 255 + 0.5), floor(clamp(b, 0, 1) * 255 + 0.5)
        local a8 = floor(clamp(a, 0, 1) * 255 + 0.5)
        local key = ((r8 * 256 + g8) * 256 + b8) * 256 + a8 + 1
        if key == d.key then return end
        d.key = key
        local off = a8 == 0
        if not d.manual then
            if off ~= d.off then
                d.off = off
                for _, h in ipairs(d.hs) do h:setNoDraw(off) end
            end
            if off then return end
        end
        local col = Color(r8, g8, b8, a8)
        for _, h in ipairs(d.hs) do h:setColor(col) end
        d.m:setVector("$color", Vector(r8 / 255, g8 / 255, b8 / 255))
        d.m:setFloat("$alpha", a8 / 255)
    end

    -- Translucent entities are sorted by the centre of their render bounds along the view direction.
    -- Stretching the bounds lets the sky draw first (large key) and the darkening veil last (negative key).
    local function forceSort(h, key, atEye)
        local f = View.fwd * (2 * key)
        local x0, y0, z0, x1, y1, z1 = min(f.x, 0), min(f.y, 0), min(f.z, 0), max(f.x, 0), max(f.y, 0), max(f.z, 0)
        if not atEye then
            local e = h:worldToLocal(View.pos)
            x0, y0, z0, x1, y1, z1 = min(x0, e.x), min(y0, e.y), min(z0, e.z), max(x1, e.x), max(y1, e.y), max(z1, e.z)
        end
        h:setRenderBounds(Vector(x0 - 128, y0 - 128, z0 - 128), Vector(x1 + 128, y1 + 128, z1 + 128))
    end

    -- Draws a hologram far from its entity. The entity stays beside the viewer, where a position is
    -- always inside the map, and the render matrix carries the mesh out to where it belongs.
    local function placeRemote(h, pos, ang, scale)
        local rel = pos - View.pos
        local m = Matrix()
        m:translate(rel)
        m:rotate(ang)
        m:scale(scale)
        h:setPos(View.pos)
        h:setRenderMatrix(m)
        forceSort(h, max(64, rel:dot(View.fwd)), true)
    end

    -- Brightness for the storm's own pieces: vertex colours are authored `gain` times brighter than
    -- they should look, which leaves room for lightning and for undoing the veil's darkening.
    local function shade(gain) return min(1, E.comp * (1 + 2 * E.flash) / gain) end

    -- Angle of an orthonormal frame (forward, left, up)
    local function basisAngle(f, l, u)
        local xy = sqrt(f.x * f.x + f.y * f.y)
        if xy > 0.001 then
            return Angle(deg(atan2(-f.z, xy)), deg(atan2(f.y, f.x)), deg(atan2(l.z, u.z)))
        end
        return Angle(deg(atan2(-f.z, xy)), deg(atan2(-l.x, l.y)), 0)
    end

    -- Orientation whose forward axis is f, spun by `spin` radians around it without ever snapping.
    local function spinAngle(f, spin)
        local rx, ry, rz = 1 - f.x * f.x, -f.x * f.y, -f.x * f.z
        local n = sqrt(rx * rx + ry * ry + rz * rz)
        if n < 0.01 then
            rx, ry, rz = -f.y * f.x, 1 - f.y * f.y, -f.y * f.z
            n = sqrt(rx * rx + ry * ry + rz * rz)
        end
        local l0 = Vector(rx / n, ry / n, rz / n)
        local m0 = f:cross(l0)
        local l = l0 * cos(spin) + m0 * sin(spin)
        return basisAngle(f, l, f:cross(l))
    end

    function Geo.quad(vs, p1, p2, p3, p4, c1, c2, c3, c4, u0, v0, u1, v1)
        local a = { pos = p1, u = u0, v = v0, color = c1 }
        local c = { pos = p3, u = u1, v = v1, color = c3 }
        local n = #vs
        vs[n + 1] = a
        vs[n + 2] = { pos = p2, u = u1, v = v0, color = c2 }
        vs[n + 3] = c
        vs[n + 4] = a
        vs[n + 5] = c
        vs[n + 6] = { pos = p4, u = u0, v = v1, color = c4 }
    end

    -- A soft sprite quad in the plane (ax, ay)
    function Geo.puff(vs, c, ax, ay, hx, hy, roll, col)
        local cs, sn = cos(roll), sin(roll)
        local dx = ax * (cs * hx) + ay * (sn * hx)
        local dy = ay * (cs * hy) - ax * (sn * hy)
        Geo.quad(vs, c - dx - dy, c + dx - dy, c + dx + dy, c - dx + dy, col, col, col, col, 0, 0, 1, 1)
    end

    function Geo.grad(stops, x)
        local a = stops[1]
        if x <= a[1] then return Color(a[2], a[3], a[4], a[5]) end
        for i = 2, #stops do
            local b = stops[i]
            if x <= b[1] then
                local t = (x - a[1]) / (b[1] - a[1])
                return Color(lerp(a[2], b[2], t), lerp(a[3], b[3], t), lerp(a[4], b[4], t), lerp(a[5], b[5], t))
            end
            a = b
        end
        return Color(a[2], a[3], a[4], a[5])
    end

    -- Sphere seen from the inside, coloured by elevation (degrees). elTop trims the upper rings.
    function Geo.sphere(radius, segs, rings, stops, elTop)
        local vs = {}
        local function point(i, j)
            local el = -pi / 2 + pi * i / rings
            local az = 2 * pi * j / segs
            return Vector(cos(el) * cos(az), cos(el) * sin(az), sin(el)) * radius, Geo.grad(stops, deg(el))
        end
        for i = 0, rings - 1 do
            if not elTop or (-90 + 180 * i / rings) < elTop then
                for j = 0, segs - 1 do
                    local p1, c1 = point(i, j)
                    local p2, c2 = point(i, j + 1)
                    local p3, c3 = point(i + 1, j + 1)
                    local p4, c4 = point(i + 1, j)
                    Geo.quad(vs, p1, p2, p3, p4, c1, c2, c3, c4, 0, 0, 1, 1)
                end
            end
        end
        return vs
    end

    -- Unit glow disc in the YZ plane: bright centre fading to nothing at the rim.
    function Geo.disc(r, g, b)
        local vs, n = {}, 22
        local c0, c1, c2 = Color(r, g, b, 255), Color(r, g, b, 105), Color(r, g, b, 0)
        local o = Vector(0, 0, 0)
        for i = 0, n - 1 do
            local a0, a1 = 2 * pi * i / n, 2 * pi * (i + 1) / n
            local p0, p1 = Vector(0, cos(a0) * 0.42, sin(a0) * 0.42), Vector(0, cos(a1) * 0.42, sin(a1) * 0.42)
            local q0, q1 = Vector(0, cos(a0), sin(a0)), Vector(0, cos(a1), sin(a1))
            vs[#vs + 1] = { pos = o, u = 0.5, v = 0.5, color = c0 }
            vs[#vs + 1] = { pos = p0, u = 0.5, v = 0.5, color = c1 }
            vs[#vs + 1] = { pos = p1, u = 0.5, v = 0.5, color = c1 }
            Geo.quad(vs, p0, q0, q1, p1, c1, c2, c2, c1, 0.5, 0.5, 0.5, 0.5)
        end
        return vs
    end

    local function pick(list) return list[random(#list)] end

    -- Plays once. Sounds that must never be missed come from a small pool of sound objects on the
    -- local player; the rest are emitted from a hidden hologram (`second` uses another one so the
    -- same file can overlap itself).
    function Au.shot(path, vol, pitch, second)
        local a = St.audio
        if not a then return end
        local v, p = clamp((vol or 1) * CONFIG.VOLUME, 0, 1), clamp(pitch or 100, 30, 255)
        local snd = not second and a.pool[path]
        if snd then
            pcall(function()
                snd:stop()
                snd:play()
                snd:setVolume(v, 0)
                snd:setPitch(p, 0)
            end)
            return
        end
        local ent = second and a.e2 or a.e1
        -- level 0 = no distance falloff; channel 6 (static) so shots overlap instead of cutting each other
        pcall(ent.emitSound, ent, path, 0, p, v, 6)
    end

    -- Thunder takes the four recordings in turn, so two peals rarely cut each other off
    function Au.thunder(vol, pitch)
        local a = St.audio
        if not a then return end
        a.peal = a.peal % #SND.thunder + 1
        Au.shot(SND.thunder[a.peal], vol, pitch)
    end

    function Au.later(delay, fn)
        local a = St.audio
        if a then a.pending[#a.pending + 1] = { t = timer.curtime() + delay, fn = fn } end
    end

    function Au.want(key, vol, pitch)
        local l = St.audio.loops[key]
        if l then l.want, l.pitch = vol, pitch end
    end

    function Au.build(q)
        q[#q + 1] = function()
            local e1, e2 = newHolo(), newHolo()

            e2:setNoDraw(true)

            local path = SND.laugh[1]
            for _, p in ipairs(SND.laugh) do
                local ok, yes = pcall(sound.exists, p)
                if ok and yes then path = p break end
            end
            St.audio.laugh = path
        end
        local function object(path, store)
            q[#q + 1] = function()
                if not sound.canCreate() then return false end
                local a = St.audio
                local file = path or a.laugh
                local ok, snd = pcall(sound.create, me, file)
                if ok and snd then
                    Res.sounds[#Res.sounds + 1] = snd
                    store(a, snd, file)
                end
            end
        end

            object(path, function(a, snd) a.loops[key] = { snd = snd, want = 0, pitch = 100, on = false, check = 0 } end)
        end
        local function pooled(a, snd, file) a.pool[file] = snd end
        for _, path in ipairs(SND.thunder) do object(path, pooled) end
        object(nil, pooled)
        for _, path in ipairs(SND.crack) do object(path, pooled) end
        object("ambient/wind/windgust_strong.wav", pooled)
        object("npc/strider/striderx_alert2.wav", pooled)
        -- Optional: a silent copy of the laugh played through bass, only to read its loudness for the jaw.

    function Au.update(now)
        local a = St.audio
        if not a then return end
        a.e1:setPos(View.pos)
        a.e2:setPos(View.pos)
        for i = #a.pending, 1, -1 do
            local p = a.pending[i]
            if now >= p.t then
                table.remove(a.pending, i)
                p.fn()
            end
        end
        if now < a.next then return end
        a.next = now + 0.16

        local wind, storm, life, shelter = E.wind, E.storm, E.life, View.open and 1 or 0.45
        Au.want("wind", max((0.1 + 0.8 * wind) * E.dark * shelter, 0.9 * E.torVol), 86 + 32 * wind - 24 * E.torNear)
        Au.want("roar", max(E.torVol, 0.5 * E.blow, 0.3 * storm * shelter), 60 + 14 * E.torNear + 30 * E.blow)
        Au.want("rumble", 0.55 * storm, 90)
        Au.want("drone", 0.4 * E.dark, 72)
        Au.want("choir", 0.2 * E.face, 58)
        Au.want("rain", (CONFIG.RAIN and 0.28 * E.rain or 0) * shelter, 128)
        Au.want("deep", E.torVol * 0.8, 64)

        for _, l in pairs(a.loops) do
            local v = clamp(l.want * CONFIG.VOLUME, 0, 1)
            if v < 0.012 then
                if l.on then
                    l.on = false
                    l.snd:stop(0.6)
                end
            else
                -- not every file loops by itself, so restart the ones that ran out
                if not l.on or (now > l.check and not l.snd:isPlaying()) then
                    l.on = true
                    l.snd:play()
                end
                if now > l.check then l.check = now + 2 end
                l.snd:setVolume(v, 0.2)
                l.snd:setPitch(l.pitch, 0.2)
            end
        end

        if wind > 0.3 and now >= a.gust then
            a.gust = now + (5 + random() * 9) / (0.4 + wind)
            Au.shot(pick(SND.gust), (0.25 + 0.5 * wind) * shelter, 78 + random() * 30)
        end
        if storm > 0.1 and now >= a.boom then
            a.boom = now + 14 + random() * 18
            Au.shot(pick(SND.boom), 0.3 + 0.3 * random(), 58 + random() * 30, true)
        end
        if E.dark > 0.6 and now >= a.moan then
            a.moan = now + 25 + random() * 30
            Au.shot(pick(SND.moan), 0.35 * life, 70 + random() * 20, true)
        end
        if E.face > 0.5 and now >= a.voice then
            a.voice = now + 16 + random() * 18
            Au.shot(pick(SND.voice), 0.5, 42 + random() * 16, true)
        end
        if E.torNear > 0.15 and now >= a.wreck then
            a.wreck = now + 1.2 + random() * 3.5 * (1.2 - E.torNear)
            Au.shot(pick(SND.wreck), 0.3 + 0.6 * E.torNear, 70 + random() * 40)
        end
    end

    -- One of three bands of a flat cloud deck, every puff turned to face the viewer below it.
    -- The bands split the deck along the direction the storm arrives from.
    function Sky.deckVerts(seed, deck, band, upX, upY)
        local rng, vs = RNG(seed), {}
        local count = floor(deck[1] * QUAL)
        local height, reach = deck[2] * SKY, deck[3] * SKY
        for _ = 1, count do
            local a, r = rng(0, 2 * pi), sqrt(rng()) * reach
            local x, y, z = cos(a) * r, sin(a) * r, height * rng(0.92, 1.1)
            local half = SKY * rng(deck[4], deck[5]) * (0.75 + 0.6 * r / reach)
            local roll, shade, alpha = rng(0, 2 * pi), rng(0.72, 1.25), rng(deck[9], deck[10])
            local w = (x * upX + y * upY) / reach
            local b = w > 0.3 and 1 or (w > -0.25 and 2 or 3)
            if band == 0 or b == band then
                local c = Vector(x, y, z)
                local f = c:getNormalized()
                local right = Vector(-f.y, f.x, 0)
                if right:getLengthSqr() < 0.0001 then right = Vector(1, 0, 0) end
                right = right:getNormalized()
                local col = Color(min(255, deck[6] * shade), min(255, deck[7] * shade), min(255, deck[8] * shade), alpha)
                Geo.puff(vs, c, right, f:cross(right), half, half, roll, col)
            end
        end
        return vs
    end

    -- Towering banks that close the horizon
    function Sky.bankVerts(seed, band, upX, upY, vs)
        local rng = RNG(seed)
        for i = 1, 56 do
            local a = (i + rng(-0.4, 0.4)) / 56 * 2 * pi
            local el, d = rad(rng(-3, 11)), SKY * rng(0.9, 0.97)
            local half, roll, shade = SKY * rng(0.14, 0.24), rng(0, 2 * pi), rng(0.7, 1.2)
            local w = cos(a) * upX + sin(a) * upY
            local b = w > 0.3 and 1 or (w > -0.25 and 2 or 3)
            if b == band then
                local f = Vector(cos(el) * cos(a), cos(el) * sin(a), sin(el))
                local right = Vector(-sin(a), cos(a), 0)
                Geo.puff(vs, f * d, right, f:cross(right), half, half, roll, Color(44 * shade, 56 * shade, 104 * shade, rng(200, 250)))
            end
        end
    end

            St.sky.decks[#St.sky.decks + 1] = { d = drawable({ h }, mat), h = h, name = "low", band = 0, key = 44000 }
        end

    function Sky.update(now)
        local sk = St.sky
        if not sk then return end
        local A, F, t, life = View.pos, E.flash, E.t, E.life

        -- The veil darkens everything on screen, the storm's own pieces included, so those are
        -- brightened by the same amount: the sky looks the same however dark the world is set.
        local veil = clamp(CONFIG.DARKNESS, 0, 0.85) * E.dark * (1 - 0.55 * F)
        E.comp = 1 / (1 - veil)
        if sk.veil then
            sk.veilH:setPos(A)
            forceSort(sk.veilH, -60000, true)
            local k = F * 0.7
            tint(sk.veil, lerp(0.022, 0.20, k), lerp(0.03, 0.25, k), lerp(0.07, 0.36, k), veil)
        end
        if sk.dome then
            sk.domeH:setPos(A)
            forceSort(sk.domeH, 60000, true)
            local m = shade(4)
            tint(sk.dome, m, m, m, E.sky)
        end
        if sk.haze then
            for i, h in ipairs(sk.haze.hs) do
                h:setPos(A)
                forceSort(h, 57000 - i * 500, true)
            end
            -- the same colour as the dome's horizon, so distant ground dissolves into the sky
            local k = E.comp * (1 + 2 * F)
            tint(sk.haze, 0.086 * k, 0.118 * k, 0.204 * k, (0.24 + 0.1 * E.rain) * E.dark)
        end

        -- the decks turn slowly around the zenith once they have closed in
        local x = t - 70
        local turn = x < 0 and 0 or (x < 30 and x * x / 60 or x - 15)
        for _, dk in ipairs(sk.decks) do
            local a, speed, drop
            if dk.name == "high" then
                a, speed, drop = ramp(t, 6 + 14 * dk.band, 46 + 14 * dk.band), 0.35, 0.08
            elseif dk.name == "mid" then
                a, speed, drop = ramp(t, 30 + 16 * dk.band, 80 + 16 * dk.band), -0.6, 0.1
            else
                a, speed, drop = ramp(t, 70, 125), 1.1, 0.07
            end
            a = a * life
            -- each deck sinks into place as it thickens
            dk.h:setPos(A + Vector(0, 0, (1 - a) * drop * SKY))
            dk.h:setAngles(Angle(1.2 * sin(now * 0.05 + dk.key), turn * speed, 1.2 * cos(now * 0.043 + dk.key)))
            forceSort(dk.h, dk.key)
            local m = shade(4)
            tint(dk.d, m, m, m, a)
        end
    end

    -- A forked bolt one unit tall, striking at the origin. Two crossed ribbons so it reads from any side.
    function Bolt.verts(seed)
        local rng, vs = RNG(seed), {}
        local axes = { Vector(1, 0, 0), Vector(0, 1, 0) }
        local function ribbon(pts, core, halo, bright)
            local cc, ce = Color(255, 255, 255, 255 * bright), Color(205, 225, 255, 0)
            local hc, he = Color(125, 165, 255, 95 * bright), Color(125, 165, 255, 0)
            for i = 1, #pts - 1 do
                local a, b = pts[i], pts[i + 1]
                for _, ax in ipairs(axes) do
                    local h, c = ax * halo, ax * core
                    Geo.quad(vs, a - h, a, b, b - h, he, hc, hc, he, 0, 0, 1, 1)
                    Geo.quad(vs, a, a + h, b + h, b, hc, he, he, hc, 0, 0, 1, 1)
                    Geo.quad(vs, a - c, a, b, b - c, ce, cc, cc, ce, 0, 0, 1, 1)
                    Geo.quad(vs, a, a + c, b + c, b, cc, ce, ce, cc, 0, 0, 1, 1)
                end
            end
        end
        local pts, disp = { Vector(rng(-0.1, 0.1), rng(-0.1, 0.1), 1), Vector(0, 0, 0) }, 0.085
        for _ = 1, 5 do
            local out = {}
            for i = 1, #pts - 1 do
                local a, b = pts[i], pts[i + 1]
                out[#out + 1] = a
                out[#out + 1] = Vector((a.x + b.x) / 2 + rng(-disp, disp), (a.y + b.y) / 2 + rng(-disp, disp), (a.z + b.z) / 2 + rng(-disp, disp) * 0.3)
            end
            out[#out + 1] = pts[#pts]
            pts, disp = out, disp * 0.55
        end
        ribbon(pts, 0.006, 0.028, 1)
        for _ = 1, floor(rng(2, 4.99)) do
            local p = pts[floor(rng(4, #pts * 0.7))]
            local dir = Vector(rng(-1, 1), rng(-1, 1), rng(-1.2, -0.3)):getNormalized()
            local len, branch = rng(0.12, 0.3), { p }
            for _ = 1, 7 do
                p = p + dir * (len / 7) + Vector(rng(-1, 1), rng(-1, 1), rng(-1, 1)) * (len * 0.05)
                branch[#branch + 1] = p
            end
            ribbon(branch, 0.003, 0.014, 0.55)
        end
        return vs
    end

    -- kind 0: sheet lightning at (azimuth, elevation). kind 1: a bolt hitting the ground at v.
    function Bolt.spawn(kind, v, seed, power)
        local b = St.bolt
        if not b or not b.ready or S.phase == 0 then return end
        local now, rng = timer.curtime(), RNG(seed + 1)
        local fl = { t0 = now, kind = kind, power = power, pulses = {}, life = 0 }
        local t = 0
        for i = 1, 1 + floor(rng(0, 3.99)) do
            fl.pulses[i] = { t = t, amp = rng(0.5, 1), decay = rng(0.05, 0.16) }
            t = t + rng(0.06, 0.22)
        end
        fl.pulses[#fl.pulses].decay = rng(0.2, 0.32)
        fl.pulses[#fl.pulses].amp = 1
        fl.life = t + 1.1

        b.gi = b.gi % 2 + 1
        fl.glow = b.glows[b.gi]
        local delay, vol, pitch
        if kind == 1 then
            local dist = (v - View.pos):getLength()
            local near = clamp(1 - dist / 9000, 0, 1)
            local height = SKY * 0.34
            b.bi = b.bi % 2 + 1
            fl.bolt = b.bolts[b.bi]
            fl.bolt.h:setMesh(b.meshes[seed % 4 + 1] or b.meshes[1])
            fl.bolt.h:setPos(v)
            fl.bolt.h:setAngles(Angle(0, seed % 360, 0))
            fl.bolt.h:setScale(Vector(height, height, height))
            fl.bolt.h:setRenderBounds(Vector(-height, -height, -height), Vector(height, height, height))
            fl.pos, fl.size, fl.weight = v + Vector(0, 0, height * 0.9), height * 0.95, 0.45 + 0.55 * near
            delay, vol, pitch = clamp(dist / 5000, 0.05, 4.5), 0.45 + 0.55 * near, 80 + 25 * near
            if near > 0.3 then pcall(trace.decal, "Scorch", v + Vector(0, 0, 32), v - Vector(0, 0, 64)) end
            if near > 0.55 then
                Au.later(delay * 0.5, function() Au.shot(pick(SND.crack), near, 85 + random() * 20) end)
                Au.later(delay * 0.5, function() Hud.shake(9 * near * near, 9, 0.7) end)
            end
        else
            local az, el = rad(v.x), rad(v.y)
            fl.dir = Vector(cos(el) * cos(az), cos(el) * sin(az), sin(el))
            fl.size, fl.weight = SKY * lerp(0.2, 0.44, power), lerp(0.2, 0.75, power) * (0.45 + 0.55 * sin(el))
            delay, vol, pitch = lerp(5.5, 1.2, power) * (0.7 + 0.6 * rng()), lerp(0.25, 0.9, power), lerp(62, 96, power)
        end
        Au.later(delay, function() Au.thunder(vol, pitch + random() * 8) end)
        b.live[#b.live + 1] = fl
    end

    function Bolt.update(now)
        local b = St.bolt
        E.flash = 0
        if not b then return end
        local best = 0
        for i = #b.live, 1, -1 do
            local fl = b.live[i]
            local dt, v = now - fl.t0, 0
            for _, p in ipairs(fl.pulses) do
                local x = dt - p.t
                if x >= 0 then v = max(v, p.amp * exp(-x / p.decay)) end
            end
            if dt > fl.life then
                v = 0
                table.remove(b.live, i)
            end
            local pos = fl.pos or (View.pos + fl.dir * (SKY * 0.78))
            placeRemote(fl.glow.h, pos, (View.pos - pos):getAngle(), Vector(1, fl.size, fl.size))
            tint(fl.glow.d, 1, 1, 1, v * (0.5 + 0.5 * fl.power))
            if fl.bolt then tint(fl.bolt.d, 1, 1, 1, v) end
            best = max(best, v * fl.weight)
        end
        E.flash = best * E.life
    end

    -- Reads where the eyes are from the model itself, then prepares the flex controllers.
    function Face.init(f)
        local h = f.h
        f.tries = f.tries + 1
        local eye
        local id = h:lookupAttachment("eyes")
        if id and id > 0 then
            local p = h:getAttachment(id)
            if p then eye = p - h:getPos() end
        end
        if not eye and f.tries < 15 then return false end
        if not eye then
            local bone = h:lookupBone("ValveBiped.Bip01_Head1")
            local p = bone and h:getBonePosition(bone)
            if p then eye = p - h:getPos() + Vector(3.2, 0, 3.4) end
        end
        if not eye or eye.z < 30 or eye.z > 110 or abs(eye.x) > 25 or abs(eye.y) > 10 then eye = Vector(3.3, 0, 67.4) end
        f.pivot = eye + Vector(-3, 0, 0)
        f.mouth = Vector(3.7, 0, -3.2) * f.unit
        f.clip = Vector(0, 0, -6.9) * f.unit
        local ok, flexes = pcall(h.getFlexes, h)
        if ok and flexes then
            for name, list in pairs(FLEX) do
                local out = {}
                for _, flex in ipairs(list) do
                    local fid = flexes[flex]
                    if fid then
                        local okb, lo, hi = pcall(h.getFlexBounds, h, fid)
                        if okb then out[#out + 1] = { fid, lo or 0, hi or 1 } end
                    end
                end
                f.chan[name], f.cur[name] = out, 0
            end
        end
        f.ready = true
        return true
    end

    function Face.flex(f, name, value)
        local list = f.chan[name]
        if not list then return end
        if abs(value - (f.last[name] or -1)) < 0.006 then return end
        f.last[name] = value
        for _, c in ipairs(list) do f.h:setFlexWeight(c[1], c[2] + (c[3] - c[2]) * value) end
    end

    function Face.hide(f)
        if f.off then return end
        f.off = true
        f.h:setNoDraw(true)
        if f.collar then tint(f.collar, 1, 1, 1, 0) end
        if f.breath then tint(f.breath, 1, 1, 1, 0) end
    end

    -- Things that happen once at a given moment of the current action
    function Face.cue(f, a, at, now)
        local fired = a.fired
        if now - a.t0 > at + 1.5 then fired[at] = true end
        if fired[at] then return false end
        if now - a.t0 < at then return false end
        fired[at] = true
        return true
    end

    function Face.update(now, dt)
        local f = St.face
        if not f then return end
        E.blow = 0
        if not f.ready and not Face.init(f) then return end

        -- the glow that gathers behind the clouds before anything can be seen
        local A, F, vis = View.pos, E.flash, E.face
        local yaw = rad(S.faceYaw)
        if f.back then
            local dir = Vector(cos(0.52) * cos(yaw), cos(0.52) * sin(yaw), sin(0.52))
            local pos = A + dir * (SKY * 0.82)
            placeRemote(f.backH, pos, (A - pos):getAngle(), Vector(1, SKY * 0.42, SKY * 0.42))
            local glow = ramp(E.t, T.FACE - 45, T.FACE + 15) * E.life
            tint(f.back, 0.42, 0.55, 0.86, glow * (0.16 + 0.2 * vis + 0.5 * F + 0.03 * sin(now * 0.7)))
        end
        if vis < 0.002 then Face.hide(f) return end
        if f.off then
            f.off = false
            f.h:setNoDraw(false)
        end
        if vis > 0.03 and not f.announced then
            f.announced = true
            Au.shot("npc/strider/striderx_alert2.wav", 0.9, 38)
            Au.shot("ambient/atmosphere/terrain_rumble1.wav", 0.9, 70)
            Hud.shake(3, 3, 4)
        end

        -- what it is doing
        if Net.face then
            f.act, Net.face = Net.face, nil
            f.act.fired = {}
            f.blinkT = now
        end
        local a = f.act
        local at = a and (now - a.t0) or 0
        if a and (at > a.dur or at < -2) then
            a, f.act = nil, nil
        end
        local kind = a and a.kind or 0
        if kind == 3 and a.laughEnd == nil then
            local okd, d = pcall(sound.duration, St.audio and St.audio.laugh or SND.laugh[1])
            d = (okd and d and d > 0.5 and d < 20) and d or 3.6
            a.pitch = 54
            a.laughEnd = min(0.4 + d * 100 / a.pitch, a.dur - 0.9)
        end
        f.wStare = approach(f.wStare, kind == 2 and 1 or 0, 2.2, dt)
        f.wLaugh = approach(f.wLaugh, (kind == 3 and at > 0.3 and at < a.laughEnd) and 1 or 0, 6, dt)
        f.wIn = approach(f.wIn, (kind == 4 and at < BLOW_INHALE) and 1 or 0, 3, dt)
        local blowing = kind == 4 and at >= BLOW_INHALE and at < a.dur - 1.3
        f.wBlow = approach(f.wBlow, blowing and 1 or 0, 5, dt)

        -- where it is: far back inside the deck at first, then pushing forward through it
        local el = rad(lerp(41, 29, vis) + 0.8 * sin(now * 0.31))
        local dist = SKY * (lerp(0.9, 0.5, vis) - 0.045 * f.wStare * ramp(at, 0, 6) - 0.035 * f.wBlow + 0.02 * f.wIn + 0.004 * sin(now * 0.5))
        local dir = Vector(cos(el) * cos(yaw), cos(el) * sin(yaw), sin(el))
        local head = A + dir * dist
        local baseAng = (-dir):getAngle()

        -- where it looks
        local P
        if kind == 1 then
            P = a.target
        elseif kind == 2 or kind == 3 then
            local ent = a.ent > 0 and entity(a.ent) or nil
            if ent and ent == me then
                P = A
            elseif ent and isValid(ent) then
                P = ent:getPos() + Vector(0, 0, 56)
            else
                P = kind == 2 and a.target or A
            end
        elseif kind == 4 then
            P = (St.tor and St.tor.pos) or a.target
        else
            P = Vector(S.centre.x + sin(now * 0.13) * S.radius * 0.6, S.centre.y + cos(now * 0.097) * S.radius * 0.6, S.centre.z)
        end
        f.gaze = P
        local lp = worldToLocal(P, Angle(0, 0, 0), head, baseAng)
        local wantYaw = clamp(deg(atan2(lp.y, lp.x)), -58, 58)
        local wantPitch = clamp(-deg(atan2(lp.z, sqrt(lp.x * lp.x + lp.y * lp.y))), -34, 38)
        -- the head follows slowly, the eyes get there first
        local headRate = kind == 2 and 2 or 1.3
        f.hy = approach(f.hy, clamp(wantYaw * 0.62, -34, 34), headRate, dt)
        f.hp = approach(f.hp, clamp(wantPitch * 0.62, -22, 24), headRate, dt)
        if now >= f.saccade then
            f.saccade = now + 0.4 + random() * 1.2
            f.sy, f.sp = (random() - 0.5) * 4.4, (random() - 0.5) * 2.8
            if kind == 2 then f.sy, f.sp = f.sy * 0.2, f.sp * 0.2 end
        end
        f.ey = approach(f.ey, clamp(wantYaw - f.hy, -30, 30) + f.sy, 26, dt)
        f.ep = approach(f.ep, clamp(wantPitch - f.hp, -22, 24) + f.sp, 26, dt)

        -- laughter: the jaw follows the loudness of the sound when it can be measured
        local env = 0
        if kind == 3 and at > 0.3 and at < a.laughEnd then
            local x = at - 0.3
            local meter = St.audio and St.audio.meter
            if meter and a.metered and not a.deaf then
                local okl, l, r = pcall(meter.getLevels, meter)
                local lvl = okl and max(l or 0, r or 0) or 0
                a.peak = max(a.peak or 0, lvl)
                if x > 0.9 and a.peak < 0.01 then a.deaf = true end
                env = clamp(lvl * 3.2, 0, 1)
            end
            if not (meter and a.metered) or a.deaf then
                env = (0.3 + 0.7 * abs(sin(pi * 5 * a.pitch / 100 * x)) ^ 1.4) * (1 - 0.4 * ramp(x, a.laughEnd - 2, a.laughEnd))
            end
        end
        f.env = approach(f.env, env, 30, dt)

        local localAng = Angle(
            f.hp - 9 * f.wLaugh - 3.5 * f.wLaugh * f.env - 11 * f.wIn + 5 * f.wBlow + 1.4 * sin(now * 0.17),
            f.hy + 2.4 * sin(now * 0.21 + 1) + 0.8 * f.wBlow * sin(now * 9),
            1.2 * sin(now * 0.13 + 2) + 2.2 * f.wLaugh * sin(now * 4.3))
        local _, total = localToWorld(Vector(0, 0, 0), localAng, Vector(0, 0, 0), baseAng)

        -- The entity itself stays beside the viewer so it is lit like the world around them and keeps
        -- its full-detail model; the render matrix carries the head out into the sky.
        local L = A + Vector(0, 0, clamp(View.ceil - 14, 0, 40))
        local rel = head - L
        local m = Matrix()
        m:translate(rel)
        m:rotate(total)
        m:scale(Vector(f.unit, f.unit, f.unit))
        m:translate(-f.pivot)
        f.h:setPos(L)
        f.h:setRenderMatrix(m)
        local e = f.hr * 2.4
        f.h:setRenderBounds(Vector(min(rel.x, 0) - e, min(rel.y, 0) - e, min(rel.z, 0) - e), Vector(max(rel.x, 0) + e, max(rel.y, 0) + e, max(rel.z, 0) + e))
        -- cut the body away below the jaw
        f.h:setClip(1, true, localToWorld(f.clip, Angle(0, 0, 0), head, total), total:getUp())

        -- a dark shape at first, then cold and pale; lightning lights it like everything else
        local c = lerp(0.1, 0.74, ramp(vis, 0.15, 1)) + 0.26 * F
        local col = floor(clamp(c, 0, 1) * 255)
        if col ~= f.col then
            f.col = col
            f.h:setColor(Color(floor(col * 0.9), floor(col * 0.95), min(255, floor(col * 1.06)), 255))
        end

        -- expression
        local wIdle = clamp(1 - f.wStare - f.wLaugh - f.wIn - f.wBlow, 0, 1)
        for name in pairs(f.chan) do
            local v = (POSE.idle[name] or 0) * wIdle + (POSE.stare[name] or 0) * f.wStare + (POSE.laugh[name] or 0) * f.wLaugh
                + (POSE.inhale[name] or 0) * f.wIn + (POSE.blow[name] or 0) * f.wBlow
            f.cur[name] = approach(f.cur[name], v, 7, dt)
        end
        local jaw = 0.02 + 0.02 * sin(now * 0.9) + 0.05 * f.wStare + (0.22 + 0.78 * f.env) * f.wLaugh + 0.6 * f.wIn + (0.13 + 0.03 * sin(now * 13)) * f.wBlow
        f.jaw = approach(f.jaw, jaw, 22, dt)
        -- blinking: rare while staring, and always on a change of attention
        if now >= f.nextBlink then
            f.blinkT = now
            f.nextBlink = now + (2.4 + random() * 4.1) * (kind == 2 and 3.5 or 1)
        end
        local bx = now - f.blinkT
        local blink = bx < 0.07 and bx / 0.07 or (bx < 0.12 and 1 or (bx < 0.25 and 1 - (bx - 0.12) / 0.13 or 0))
        blink = max(blink, 0.3 * f.wBlow, 0.25 * f.wLaugh, f.ep > 8 and 0.15 or 0)
        for name in pairs(f.chan) do
            if name == "jaw" then
                Face.flex(f, name, clamp(f.jaw, 0, 1))
            elseif name == "blink" then
                Face.flex(f, name, clamp(blink, 0, 1))
            else
                Face.flex(f, name, clamp(f.cur[name], 0, 1))
            end
        end
        if CONFIG.EYE_TRACKING then
            local key = floor(f.ey * 5) .. ":" .. floor(f.ep * 5)
            if key ~= f.eyeKey then
                f.eyeKey = key
                -- for non-NPC models the eye target is relative to the eyes attachment
                pcall(f.h.setEyeTarget, f.h, Vector(4000, 4000 * tan(rad(f.ey)), -4000 * tan(rad(f.ep))))
            end
        end

        -- cloud around the head
        if f.collar then
            placeRemote(f.collarH, head, baseAng, Vector(1, 1, 1))
            local m2 = shade(4)
            tint(f.collar, m2, m2, m2, ramp(vis, 0.02, 0.35) * 0.96)
        end

        -- sounds and the breath
        if kind == 2 and Face.cue(f, a, 0.2, now) then
            Au.shot(pick(SND.voice), 0.6, 44, true)
        end
        if kind == 3 and Face.cue(f, a, 0.3, now) then
            local au = St.audio
            if au then
                Au.shot(au.laugh, 1, a.pitch)
                -- a second, slightly lower copy a moment later makes one voice into something much larger
                Au.later(0.11, function() Au.shot(au.laugh, 0.65, a.pitch - 4, true) end)
                Hud.shake(1.6, 3, a.laughEnd)
                if au.meter then
                    a.metered = pcall(function()
                        au.meter:setTime(0)
                        au.meter:setPitch(a.pitch / 100)
                        au.meter:setVolume(0)
                        au.meter:play()
                    end)
                end
            end
        end
        if kind == 3 and a.metered and at > a.laughEnd and not a.paused then
            a.paused = true
            pcall(St.audio.meter.pause, St.audio.meter)
        end
        if kind == 4 then
            if Face.cue(f, a, 0.1, now) then Au.shot("ambient/levels/labs/teleport_preblast_suckin1.wav", 0.9, 62) end
            if Face.cue(f, a, BLOW_INHALE, now) then
                Au.shot("ambient/wind/windgust_strong.wav", 1, 68)
                Au.shot("ambient/wind/windgust.wav", 1, 52, true)
                Hud.shake(4, 6, a.dur - BLOW_INHALE - 1)
            end
        end
        if f.breath then
            local grow = (kind == 4) and ramp(at, BLOW_INHALE, BLOW_INHALE + 1.8) * (1 - ramp(at, a.dur - 2.2, a.dur - 0.9)) or 0
            if grow > 0.004 then
                local mouth = localToWorld(f.mouth, Angle(0, 0, 0), head, total)
                local to = P - mouth
                local len = to:getLength()
                local reach = max(200, min(len, SKY * 2.2) * ramp(at, BLOW_INHALE, BLOW_INHALE + 1.8))
                placeRemote(f.breathH, mouth, spinAngle(to / max(len, 1), now * 2.4), Vector(reach, reach, reach))
                E.blow = grow
            end
            tint(f.breath, 0.62 + 0.3 * F, 0.68 + 0.3 * F, 0.78 + 0.22 * F, grow * (0.5 + 0.08 * sin(now * 11)))
        end
    end

    -- One slice of the funnel: axis along X from -0.5 to 0.5, radius 1. A dark core you cannot see
    -- through, a skin of soft puffs, and fins that stand out from it so the outline is never a hard edge.
    function Tor.sliceVerts(seed, p)
        local rng, vs = RNG(seed), {}
        local X = Vector(1, 0, 0)
        if p[4] > 0 then
            local sides, cr = 10, 0.62
            local dark = Color(p[1] * p[4], p[2] * p[4], p[3] * p[4], 238)
            local clear = Color(p[1] * p[4], p[2] * p[4], p[3] * p[4], 0)
            for i = 0, sides - 1 do
                local a0, a1 = 2 * pi * i / sides, 2 * pi * (i + 1) / sides
                local y0, z0, y1, z1 = cos(a0) * cr, sin(a0) * cr, cos(a1) * cr, sin(a1) * cr
                -- uv pinned to the opaque middle of the sprite gives a flat colour from the same material
                Geo.quad(vs, Vector(-0.5, y0, z0), Vector(-0.5, y1, z1), Vector(-0.3, y1, z1), Vector(-0.3, y0, z0), clear, clear, dark, dark, 0.5, 0.5, 0.5, 0.5)
                Geo.quad(vs, Vector(-0.3, y0, z0), Vector(-0.3, y1, z1), Vector(0.3, y1, z1), Vector(0.3, y0, z0), dark, dark, dark, dark, 0.5, 0.5, 0.5, 0.5)
                Geo.quad(vs, Vector(0.3, y0, z0), Vector(0.3, y1, z1), Vector(0.5, y1, z1), Vector(0.5, y0, z0), dark, dark, clear, clear, 0.5, 0.5, 0.5, 0.5)
            end
        end
        local function colour(x)
            local shade = rng(0.72, 1.2)
            local edge = 1 - ramp(abs(x), 0.3, 0.5)
            return Color(min(255, p[1] * shade), min(255, p[2] * shade), min(255, p[3] * shade), rng(p[5], p[6]) * edge)
        end
        for _ = 1, p[7] do
            local x, a, rho = rng(-0.42, 0.42), rng(0, 2 * pi), rng(0.86, 1.08)
            local tangent = Vector(0, -sin(a), cos(a))
            Geo.puff(vs, Vector(x, cos(a) * rho, sin(a) * rho), X, tangent, rng(0.3, 0.5), rng(0.5, 0.85), rng(-0.5, 0.5), colour(x))
        end
        for _ = 1, p[8] do
            local x, a = rng(-0.4, 0.4), rng(0, 2 * pi)
            local radial = Vector(0, cos(a), sin(a))
            Geo.puff(vs, radial * 1.05 + Vector(x, 0, 0), X, radial, rng(0.35, 0.5), rng(0.3, 0.45), rng(-0.6, 0.6), colour(x))
        end
        for _ = 1, p[9] do
            local x, a, rho = rng(-0.4, 0.4), rng(0, 2 * pi), rng(0.2, 0.85)
            local tangent = Vector(0, -sin(a), cos(a))
            Geo.puff(vs, Vector(x, cos(a) * rho, sin(a) * rho), X, tangent, rng(0.35, 0.5), rng(0.45, 0.8), rng(-0.5, 0.5), colour(x))
        end
        for _ = 1, p[10] do
            local x, a, rho = rng(-0.3, 0.3), rng(0, 2 * pi), sqrt(rng()) * 1.05
            local radial = Vector(0, cos(a), sin(a))
            local size = rng(0.35, 0.6)
            Geo.puff(vs, radial * rho + Vector(x, 0, 0), radial, Vector(0, -sin(a), cos(a)), size, size, rng(0, 2 * pi), colour(x * 0.5))
        end
        return vs
    end

    function Tor.build(q)
        local count = clamp(floor(14 * QUAL + 0.5), 8, 20)
        q[#q + 1] = function()
            St.torBuild = { meshes = {}, segs = {}, skirt = {}, wall = {}, debris = {} }
            St.torBuild.mat = newMat(St.tex.puff, MAT_BASE)
            St.torBuild.dustMat = newMat(St.tex.wisp, MAT_BASE)
            St.torBuild.wallMat = newMat(St.tex.puff, MAT_BASE)
        end
        for i, name in ipairs({ "base", "body", "top", "dust", "wall" }) do
            q[#q + 1] = function() St.torBuild.meshes[name] = newMesh(Tor.sliceVerts(S.seed + 100 + i, FUNNEL[name])) end
        end
        q[#q + 1] = function()
            local b = St.torBuild
            for i = 1, count do
                local u0, u1 = ((i - 1) / count) ^ 1.6, (i / count) ^ 1.6
                local mid = (u0 + u1) / 2
                local kind = mid < 0.16 and "base" or (mid < 0.6 and "body" or "top")
                b.segs[i] = { h = meshHolo(b.meshes[kind], b.mat, 600), u0 = u0, u1 = u1, mid = mid, spin = random() * 6, off = true, len = 0, r = 0, ext = 600 }
            end
        end
        q[#q + 1] = function()
            local b = St.torBuild
            for i = 1, 3 do b.skirt[i] = { h = meshHolo(b.meshes.dust, b.dustMat, 3000), spin = i * 2, len = 0, r = 0 } end
            for i = 1, 2 do b.wall[i] = { h = meshHolo(b.meshes.wall, b.wallMat, 6000), spin = i * 3, len = 0, r = 0 } end
        end
        local debris = floor(20 * QUAL)
        for i = 1, debris do
            q[#q + 1] = function()
                local def = DEBRIS[(i - 1) % #DEBRIS + 1]
                local ok, exists = pcall(game.modelExists, def[1])
                if ok and exists == false then return end
                local h = newHolo(def[1])
                h:setNoDraw(true)
                local rng = RNG(S.seed + i * 131)
                St.torBuild.debris[#St.torBuild.debris + 1] = {
                    h = h, size = def[2], s = 0, off = true, u0 = rng(0, 0.6), climb = rng(0.012, 0.035), rho = rng(1.15, 2.9),
                    th = rng(0, 6.28), bob = rng(20, 90), bf = rng(0.6, 1.8), t1 = rng(-190, 190), t2 = rng(-230, 230), t3 = rng(-150, 150),
                }
            end
        end
        q[#q + 1] = function()
            local b = St.torBuild
            local hs = {}
            for i, s in ipairs(b.segs) do hs[i] = s.h end
            St.tor = {
                segs = b.segs, skirt = b.skirt, wall = b.wall, debris = b.debris,

                dDust = drawable({ b.skirt[1].h, b.skirt[2].h, b.skirt[3].h }, b.dustMat),
                dWall = drawable({ b.wall[1].h, b.wall[2].h }, b.wallMat),
                ph1 = random() * 6, ph2 = random() * 6, I = 0, vx = 0, vy = 0, rb = 60, rt = 400, h = 2000, lagX = 0, lagY = 0, sway = 1, dust = 0, tick = 0,
            }
        end
    end

    -- The funnel's centre line: pinned to the ground point and to the cloud, snaking in between,
    -- with the top trailing behind the way it travels.
    function Tor.centreAt(tr, u, now)
        local amp = (tr.rb * 1.3 + 50) * sin(pi * min(u * 1.1, 1)) * tr.sway
        local a1 = 2 * pi * (0.9 * u - 0.07 * now) + tr.ph1
        local a2 = 2 * pi * (2.1 * u + 0.11 * now) + tr.ph2
        local lag = u ^ 1.6
        return Vector(
            tr.pos.x + tr.lagX * lag + (sin(a1) + 0.4 * sin(a2)) * amp,
            tr.pos.y + tr.lagY * lag + (cos(a1 * 0.83 + 1.3) + 0.4 * cos(a2 * 1.21)) * amp,
            tr.pos.z + u * tr.h)
    end

    function Tor.radiusAt(tr, u) return tr.rb + (tr.rt - tr.rb) * u ^ 2.4 end

    -- Moves a slice, only touching scale and bounds when they have really changed
    function Tor.place(s, pos, ang, len, r)
        s.h:setPos(pos)
        s.h:setAngles(ang)
        if abs(len - s.len) > s.len * 0.02 or abs(r - s.r) > s.r * 0.02 then
            s.len, s.r = len, r
            s.h:setScale(Vector(len, r, r))
            local ext = max(len, r) * 1.3
            if not s.ext or abs(ext - s.ext) > s.ext * 0.15 then
                s.ext = ext
                s.h:setRenderBounds(Vector(-ext, -ext, -ext), Vector(ext, ext, ext))
            end
        end
    end

    function Tor.hide(tr)
        if tr.hidden then return end
        tr.hidden = true
        for _, s in ipairs(tr.segs) do
            if not s.off then
                s.off = true
                s.h:setNoDraw(true)
            end
        end
        for _, d in ipairs(tr.debris) do
            if not d.off then
                d.off = true
                d.h:setNoDraw(true)
            end
        end
        tint(tr.dDust, 1, 1, 1, 0)
        tint(tr.dWall, 1, 1, 1, 0)
        tr.pos = nil
    end

    function Tor.update(now, dt)
        local tr, sn = St.tor, Net.tor
        E.torNear, E.torVol = 0, 0
        if not tr then return end
        if not sn then Tor.hide(tr) return end
        tr.hidden = false

        -- smooth the 5 Hz server path into continuous motion
        local ahead = clamp(now - sn.t, 0, 0.6)
        local tx, ty = sn.x + sn.vx * ahead, sn.y + sn.vy * ahead
        if not tr.pos then
            tr.pos, tr.vx, tr.vy, tr.I = Vector(tx, ty, sn.z), sn.vx, sn.vy, 0
        end
        local k = 1 - exp(-5 * dt)
        tr.pos = Vector(lerp(tr.pos.x, tx, k), lerp(tr.pos.y, ty, k), approach(tr.pos.z, sn.z, 3, dt))
        tr.vx, tr.vy = approach(tr.vx, sn.vx, 2, dt), approach(tr.vy, sn.vy, 2, dt)
        tr.I = approach(tr.I, sn.I, 1.5, dt)

        local I, F, life, age = tr.I, E.flash, E.life, now - sn.born
        local down = ramp(age, 2, 24) * life            -- how far the funnel has come down from the cloud
        tr.h = clamp(View.pos.z + 0.27 * SKY - tr.pos.z, 1500, 0.62 * SKY)
        tr.rb = lerp(55, 360, I) * (0.45 + 0.55 * down)
        tr.rt = lerp(420, 1700, I) * (0.6 + 0.4 * ramp(age, 0, 12))
        tr.sway = lerp(1.6, 0.5, I)
        local speed = sqrt(tr.vx * tr.vx + tr.vy * tr.vy)
        local lag = min(speed * 2.6, tr.h * 0.3)
        tr.lagX, tr.lagY = speed > 1 and -tr.vx / speed * lag or 0, speed > 1 and -tr.vy / speed * lag or 0
        local v0 = lerp(520, 900, I)

        local dx, dy = tr.pos.x - View.pos.x, tr.pos.y - View.pos.y
        local dist = sqrt(dx * dx + dy * dy)
        E.torNear = clamp(1 - dist / 4500, 0, 1) * sqrt(I) * down
        E.torVol = I ^ 0.6 * clamp(1.15 - dist / 10000, 0.12, 1) * ramp(age, 0, 20) * life

        -- funnel
        local lit = shade(3)
        tint(tr.dFunnel, lit, lit, lit, 0.96 * ramp(down, 0, 0.12))
        local c0 = Tor.centreAt(tr, 0, now)
        for i, s in ipairs(tr.segs) do
            local c1 = Tor.centreAt(tr, s.u1, now)
            if s.mid < 1 - down * 1.02 then
                if not s.off then
                    s.off = true
                    s.h:setNoDraw(true)
                end
            else
                if s.off then
                    s.off = false
                    s.h:setNoDraw(false)
                end
                local axis = c1 - c0
                local len = axis:getLength()
                local tip = clamp((s.mid - (1 - down)) / 0.12, 0.25, 1)
                local r = Tor.radiusAt(tr, s.mid) * tip * (1 + 0.07 * sin(now * 0.8 + i * 1.7))
                s.spin = s.spin + clamp(v0 / r, 0.3, 4.2) * dt
                Tor.place(s, (c0 + c1) / 2, spinAngle(axis / max(len, 1), s.spin), len * 1.5, r)
            end
            c0 = c1
        end

        -- debris cloud on the ground: it starts to turn before the funnel reaches it
        local up = Vector(0, 0, 1)
        local touch = ramp(down, 0.5, 0.92)
        tint(tr.dDust, lit, lit, lit, touch * 0.85)
        if touch > 0.003 then
            for j, s in ipairs(tr.skirt) do
                local hh = tr.rb * (1.7 - 0.3 * j) + 50
                s.spin = s.spin + (2.9 - 0.6 * j) * dt
                Tor.place(s, tr.pos + Vector(0, 0, hh * 0.45), spinAngle(up, s.spin), hh, tr.rb * (0.7 + 1.3 * j) + 60 * j)
            end
        end
        -- wall cloud: the lowering under the deck that the funnel hangs from
        local lowering = ramp(age, 0, 9) * life
        tint(tr.dWall, lit, lit, lit, lowering * 0.92)
        if lowering > 0.003 then
            local top = Tor.centreAt(tr, 1, now)
            for j, s in ipairs(tr.wall) do
                local wh = tr.h * 0.1
                s.spin = s.spin + (j == 1 and 0.26 or -0.13) * dt
                Tor.place(s, top - Vector(0, 0, wh * (0.3 + 0.5 * (j - 1)) * lowering), spinAngle(up, s.spin), wh, tr.rt * (0.6 + 0.9 * j))
            end
        end

        -- debris riding the wind
        tr.tick = tr.tick + 1
        local busy = cpuAverage() > cpuMax() * 0.55
        if not busy or tr.tick % 2 == 0 then
            local step = (busy and 2 or 1) * dt
            local shown = floor(#tr.debris * clamp(I * 1.5, 0, 1) * ramp(down, 0.85, 1))
            for n, d in ipairs(tr.debris) do
                if n > shown then
                    if not d.off then
                        d.off = true
                        d.h:setNoDraw(true)
                    end
                else
                    if d.off then
                        d.off = false
                        d.h:setNoDraw(false)
                    end
                    local u = (d.u0 + d.climb * age) % 0.62
                    local r = Tor.radiusAt(tr, u) * d.rho + 40
                    d.th = d.th + clamp(v0 * 0.7 / r, 0.25, 3.2) * step
                    local c = Tor.centreAt(tr, u, now)
                    d.h:setPos(Vector(c.x + cos(d.th) * r, c.y + sin(d.th) * r, c.z + d.bob * sin(now * d.bf + n)))
                    d.h:setAngles(Angle((now * d.t1) % 360, (now * d.t2) % 360, (now * d.t3) % 360))
                    -- shrink to nothing where a piece is flung out at the top and picked up again below
                    local sc = d.size * clamp(sqrt(sin(pi * u / 0.62)) * 1.6, 0.05, 1)
                    if abs(sc - d.s) > 0.04 then
                        d.s = sc
                        d.h:setScale(Vector(sc, sc, sc))
                    end
                end
            end
        end

        -- loose dust torn off the ground
        local wx = St.wx
        if wx and wx.dust and touch > 0.2 and dist < 8000 and not busy then
            tr.dust = tr.dust + 55 * (0.3 + I) * QUAL * dt
            local n = min(floor(tr.dust), 6)
            tr.dust = tr.dust - floor(tr.dust)
            for _ = 1, n do
                if wx.dust:getParticlesLeft() < 3 then break end
                local a, rr = random() * 2 * pi, tr.rb * (1.5 + random() * 2.5)
                local p = wx.dust:add(wx.puff, tr.pos + Vector(cos(a) * rr, sin(a) * rr, random() * tr.rb), 60, 260 * (0.5 + I), 0, 0, 90, 0, 1.1)
                local sp = 400 + random() * 500
                p:setVelocity(Vector(-sin(a) * sp - cos(a) * 200, cos(a) * sp - sin(a) * 200, 100 + random() * 400))
                p:setColor(Color(96, 86, 73))
                p:setRoll(random() * 6)
                p:setRollDelta(random() * 2 - 1)
            end
        end

        if E.torNear > 0.05 and now >= (tr.shake or 0) then
            tr.shake = now + 0.4
            Hud.shake(7 * E.torNear * E.torNear, 7, 0.6)
        end
    end

    -----------------------------------------------------------------------
    -- Rain and wind-blown dust
    -----------------------------------------------------------------------

    function Wx.build(q)
        q[#q + 1] = function()
            local w = { acc = 0, dustAcc = 0, flip = 1, next = 0 }
            local ok, puff = pcall(material.load, "particle/particle_smokegrenade")
            if not ok then return end
            w.puff = puff
            local okr, drop = pcall(material.load, "particle/rain")
            w.drop = okr and drop or puff
            local oke, e1 = pcall(particle.create, View.pos, false)
            if not oke then return end
            w.rain = { e1 }
            Res.emitters[#Res.emitters + 1] = e1
            for _ = 1, 2 do
                local ok2, e = pcall(particle.create, View.pos, false)
                if ok2 then
                    Res.emitters[#Res.emitters + 1] = e
                    if #w.rain < 2 then w.rain[2] = e else w.dust = e end
                end
            end
            St.wx = w
        end
    end

    function Wx.update(now, dt)
        local w = St.wx
        if not w then return end
        if cpuAverage() > cpuMax() * 0.6 then return end
        local wind = E.wind + 0.6 * E.blow
        local A = View.pos

        -- rain only falls where a short trace upwards finds open sky, so it stays out of buildings
        w.acc = w.acc + (CONFIG.RAIN and E.rain or 0) * 230 * QUAL * dt
        local n = min(floor(w.acc), 24)
        w.acc = w.acc - floor(w.acc)
        if n > 0 then
            local cols = min(4, n)
            local per = math.ceil(n / cols)
            for _ = 1, cols do
                local a, r = random() * 2 * pi, sqrt(random()) * 560
                local base = A + Vector(cos(a) * r + View.fwd.x * 160, sin(a) * r + View.fwd.y * 160, 0)
                local tr = trace.line(base, base + Vector(0, 0, 520), nil, WORLD)
                if not tr.StartSolid and (tr.HitSky or not tr.Hit) then
                    local top = tr.Hit and (tr.HitPos.z - base.z - 8) or 520
                    w.flip = w.flip % #w.rain + 1
                    local em = w.rain[w.flip]
                    for _ = 1, per do
                        if em:getParticlesLeft() < 3 then break end
                        local len = 55 + random() * 40
                        local p = em:add(w.drop, base + Vector(random() * 140 - 70, random() * 140 - 70, top * (0.55 + 0.45 * random())), 1.5, 1.5, len, len, 95, 60, 0.45)
                        local push = (300 + random() * 500) * wind
                        p:setVelocity(Vector(E.windX * push, E.windY * push, -1450))
                        p:setColor(Color(175, 190, 215))
                    end
                end
            end
        end

        -- streaks of dust and spray when the wind gets up
        if w.dust and View.open then
            w.dustAcc = w.dustAcc + (max(0, wind - 0.45) * 60 + E.blow * 50) * dt
            local m = min(floor(w.dustAcc), 5)
            w.dustAcc = w.dustAcc - floor(w.dustAcc)
            for _ = 1, m do
                if w.dust:getParticlesLeft() < 12 then break end
                local a, r = random() * 2 * pi, sqrt(random()) * 700
                local p = w.dust:add(w.puff, A + Vector(cos(a) * r, sin(a) * r, random() * 180 - 60), 30, 140, 0, 0, 34, 0, 0.9)
                local push = 500 + random() * 800
                p:setVelocity(Vector(E.windX * push, E.windY * push, random() * 120 - 30))
                p:setColor(Color(120, 112, 100))
                p:setRoll(random() * 6)
                p:setRollDelta(random() * 3 - 1.5)
            end
        end
    end

    -----------------------------------------------------------------------
    -- Lifecycle
    -----------------------------------------------------------------------

    local function destroyAll()
        for _, s in ipairs(Res.sounds) do
            pcall(s.stop, s)
            pcall(s.destroy, s)
        end
        for _, b in ipairs(Res.bass) do pcall(b.stop, b) end
        for _, e in ipairs(Res.emitters) do pcall(e.destroy, e) end
        for _, h in ipairs(Res.holos) do pcall(h.remove, h) end
        for _, m in ipairs(Res.meshes) do pcall(m.destroy, m) end
        for _, m in ipairs(Res.mats) do pcall(m.destroy, m) end
        freshResources()
        St, Net, queue, qpos, qwait, broken = {}, {}, nil, 1, 0, {}
        E.flash, E.blow, E.torNear, E.torVol = 0, 0, 0, 0
    end

    local function plan()
        local q = {}
        Sky.build(q)
        Au.build(q)
        Bolt.build(q)
        Wx.build(q)
        Face.build(q)
        Tor.build(q)
        return q
    end

    -- One small piece per frame, and only while there is CPU to spare.
    local function buildStep()
        if not queue then return end
        if cpuAverage() > cpuMax() * 0.45 then return end
        local fn = queue[qpos]
        if not fn then
            queue = nil
            return
        end
        local ok, res = pcall(fn)
        if not ok then
            if me == owner() then print("[apocalypse] build step " .. qpos .. " failed: " .. tostring(res)) end
            qpos, qwait = qpos + 1, 0
        elseif res ~= false then
            qpos, qwait = qpos + 1, 0
        else
            -- a step that is waiting for a quota gives up after a while instead of stalling the rest
            qwait = qwait + 1
            if qwait > 150 then qpos, qwait = qpos + 1, 0 end
        end
    end

    local function updateView()
        local e = eyePos()
        -- the last rendered view is one frame old: lead it by the movement of the previous frame
        if View.last and (e - View.last):getLengthSqr() < 250000 then
            View.pos = e + (e - View.last)
        else
            View.pos = e
        end
        View.last = e
        View.fwd = eyeVector()
        View.n = View.n + 1
        if View.n % 3 == 0 then
            local tr = trace.line(e, e + Vector(0, 0, 3000), nil, WORLD)
            if tr.StartSolid then
                View.open, View.ceil = false, 0
            else
                View.open = tr.HitSky or not tr.Hit
                View.ceil = tr.Hit and (tr.HitPos.z - e.z) or 3000
            end
        end
    end

    local function environment(now)
        local t = (now - S.t0) * CONFIG.TIME_SCALE
        local life, faceOut = 1, 1
        if S.phase == 2 then
            life = 1 - ramp(now - S.tEnd, 0, CONFIG.END_TIME * 0.9)
            faceOut = 1 - ramp(now - S.tEnd, 0, CONFIG.END_TIME * 0.45)
        end
        local yaw = rad(S.faceYaw)
        E.t, E.life = t, life
        E.dark = ramp(t, 0, 110) * life
        E.sky = ramp(t, 0, 85) * life
        E.storm = ramp(t, T.STORM, T.STORM_PEAK) * life
        E.rain = ramp(t, T.RAIN, T.RAIN + 55) * life
        E.face = ramp(t, T.FACE, T.FACE_OUT) * faceOut
        E.wind = clamp((0.12 + 0.55 * ramp(t, 20, 170) + 0.5 * E.blow + 0.3 * E.torNear) * life, 0, 1.3)
        -- the wind comes from the face
        E.windX, E.windY = -cos(yaw), -sin(yaw)
    end

    hook.add("think", "apoc_client", function()
        if S.phase == 0 then return end
        if cpuAverage() > cpuMax() * 0.85 then return end
        local now, dt = timer.curtime(), clamp(timer.frametime(), 0.001, 0.1)
        updateView()
        environment(now)
        buildStep()
        guard("lightning", Bolt.update, now)
        guard("sky", Sky.update, now)
        guard("face", Face.update, now, dt)
        guard("tornado", Tor.update, now, dt)
        guard("weather", Wx.update, now, dt)
        guard("audio", Au.update, now)
    end)

    net.receive("apoc_state", function()
        local phase, t0 = net.readUInt(2), net.readDouble()
        local tEnd, seed, yaw, centre, radius = net.readDouble(), net.readUInt(24), net.readFloat(), net.readVector(), net.readFloat()
        if phase == 0 or seed ~= S.seed then destroyAll() end
        S.phase, S.t0, S.tEnd, S.seed, S.faceYaw, S.centre, S.radius = phase, t0, tEnd, seed, yaw, centre, radius
        if phase ~= 0 and not queue and not St.sky then
            queue, qpos = plan(), 1
        end
    end)

    net.receive("apoc_bolt", function()
        local kind, v, seed, power = net.readUInt(2), net.readVector(), net.readUInt(16), net.readFloat()
        guard("lightning", Bolt.spawn, kind, v, seed, power)
    end)

    net.receive("apoc_face", function()
        Net.face = { kind = net.readUInt(3), t0 = net.readDouble(), dur = net.readFloat(), target = net.readVector(), ent = net.readUInt(13), seed = net.readUInt(16) }
    end)

    net.receive("apoc_tor", function()
        local sn = { t = net.readDouble(), x = net.readFloat(), y = net.readFloat(), z = net.readFloat(), vx = net.readFloat(), vy = net.readFloat(), I = net.readFloat(), born = net.readDouble() }
        if S.phase ~= 0 then Net.tor = sn end
    end)

    hook.add("removed", "apoc_cleanup", destroyAll)

    if CONFIG.HUD_EFFECTS then
        pcall(setupPermissionRequest, { "enablehud" }, "Apocalypse would like to shake your camera and flash your screen when lightning strikes.", true)
        if me == owner() then
            local ok, yes = pcall(hasPermission, "enablehud")
            if ok and yes then pcall(enableHud, nil, true) end
        end
        hook.add("permissionrequest", "apoc_perm", function()
            local ok, yes = pcall(permissionRequestSatisfied)
            if ok and yes then pcall(enableHud, nil, true) end
        end)
        hook.add("drawhud", "apoc_flash", function()
            if S.phase == 0 or E.flash < 0.03 then return end
            local w, h = render.getResolution()
            render.setRGBA(200, 220, 255, E.flash * 60)
            render.drawRect(0, 0, w, h)
        end)
    end
